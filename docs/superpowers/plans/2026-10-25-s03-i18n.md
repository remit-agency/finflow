> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The whole app behind login plus `/login` and `/register` switch between Polish (default), English and Ukrainian; money and dates follow the UI language; amount inputs understand comma and dot; the code gets a Polish business-day calendar and a Warsaw-based "today".

**Architecture:** `next-intl` 4 without i18n routing. The language comes from the `NEXT_LOCALE` cookie, else `Accept-Language`, else `pl` (`i18n/request.ts`). Pure helpers (`lib/i18n/*`, `lib/utils/amount-input.ts`, `lib/utils/pl-calendar.ts`, `lib/utils/today.ts`) carry the logic and are unit-tested. A guard test fails CI when a hard-coded Cyrillic UI string or `'uk-UA'` is left in the app code.

```ts
function flatten(tree: Tree, prefix = ''): Record<string, string> {
  const out: Record<string, string> = {}
  for (const [key, value] of Object.entries(tree)) {
    const path = prefix ? `${prefix}.${key}` : key
    if (typeof value === 'string') out[path] = value
    else Object.assign(out, flatten(value, path))
  }
  return out
}

// Argument names of an ICU message: `{name}`, `{count, plural, ...}` -> name, count.
// (`{# dzień}` inside a plural branch has no identifier, so it is not an argument.)
function icuArguments(message: string): string[] {
  const names = new Set<string>()
  for (const match of message.matchAll(/\{\s*([A-Za-z_][A-Za-z0-9_]*)\s*(?:,|\})/g)) names.add(match[1])
  return [...names].sort()
}
```

`lib/i18n/negotiate.ts`:

```ts
import { DEFAULT_LOCALE, isLocale, type Locale } from './config'

interface Candidate {
  tag: string
  q: number
}

// "pl-PL,pl;q=0.9,en;q=0.8" -> [{pl, 1}, {pl, 0.9}, {en, 0.8}] ordered by q (stable for equal q).
function parseAcceptLanguage(header: string): Candidate[] {
  const candidates: Candidate[] = []
  for (const part of header.split(',')) {
    const [rawTag, ...params] = part.trim().split(';')
    const tag = rawTag.trim().toLowerCase()
    if (!tag || tag === '*') continue
    let q = 1
    for (const param of params) {
      const [name, value] = param.trim().split('=')
      if (name?.trim().toLowerCase() === 'q') q = Number(value)
    }
    if (!Number.isFinite(q) || q <= 0) continue
    candidates.push({ tag: tag.split('-')[0], q })
  }
  return candidates.map((c, index) => ({ c, index })).sort((a, b) => b.c.q - a.c.q || a.index - b.index).map(({ c }) => c)
}

/** A valid cookie wins; otherwise the best supported browser language; otherwise Polish. */
export function resolveLocale(input: { cookie?: string | null; acceptLanguage?: string | null }): Locale {
  if (isLocale(input.cookie)) return input.cookie
  for (const { tag } of parseAcceptLanguage(input.acceptLanguage ?? '')) {
    if (isLocale(tag)) return tag
  }
  return DEFAULT_LOCALE
}
```

```ts
import { INTL_LOCALE, type Locale } from './config'

// Pure, locale-aware formatting (S03). `locale` is the UI language ('pl' | 'en' | 'uk'); the
// currency comes from the company, never from the language.

function moneyFormat(currency: string, locale: Locale) {
  return new Intl.NumberFormat(INTL_LOCALE[locale], { style: 'currency', currency })
}

export function formatMoney(amount: number, currency: string, locale: Locale): string {
  return moneyFormat(currency, locale).format(amount)
}

/**
 * `formatMoney` cut around its decimals, so a big total can show them smaller:
 * `whole` + `fraction` + `rest` is exactly the formatted string. `fraction` is empty for a
 * currency without minor units.
 */
export function splitMoney(amount: number, currency: string, locale: Locale): { whole: string; fraction: string; rest: string } {
  const parts = moneyFormat(currency, locale).formatToParts(amount)
  const decimal = parts.findIndex((part) => part.type === 'decimal')
  if (decimal === -1) return { whole: parts.map((p) => p.value).join(''), fraction: '', rest: '' }
  const fractionEnd = parts[decimal + 1]?.type === 'fraction' ? decimal + 2 : decimal + 1
  const join = (from: number, to?: number) => parts.slice(from, to).map((p) => p.value).join('')
  return { whole: join(0, decimal), fraction: join(decimal, fractionEnd), rest: join(fractionEnd) }
}

/** The sign formatMoney uses for a currency (`€`, `zł`), e.g. as an input suffix. */
export function currencySymbol(currency: string, locale: Locale): string {
  return moneyFormat(currency, locale).formatToParts(0).find((part) => part.type === 'currency')?.value ?? currency
}

export function formatNumber(value: number, locale: Locale, options?: Intl.NumberFormatOptions): string {
  return new Intl.NumberFormat(INTL_LOCALE[locale], options).format(value)
}

/**
 * Stored dates are calendar days at 00:00 UTC (see lib/utils/date.ts), so they are formatted in UTC
 * by default: the day never shifts with the viewer's time zone. Pass `timeZone` for real instants.
 */
export function formatDate(date: Date, locale: Locale, options: Intl.DateTimeFormatOptions = { dateStyle: 'medium' }): string {
  return new Intl.DateTimeFormat(INTL_LOCALE[locale], { timeZone: 'UTC', ...options }).format(date)
}

export function formatWeekday(date: Date, locale: Locale, weekday: 'short' | 'long' = 'short'): string {
  return new Intl.DateTimeFormat(INTL_LOCALE[locale], { weekday, timeZone: 'UTC' }).format(date)
}
```

`lib/utils/amount-input.ts`:

```ts
/**
 * Parses what a person types into an amount field, whatever their keyboard or habit:
 * `1 234,50`, `1234.5`, `1.234,50`, `1,234.50`, `0,125`. Returns null for anything else
 * (empty, letters, a minus sign, `1e3`, several decimal marks). Only non-negative amounts exist
 * in the forms, so a sign is rejected. Pure, independent of the UI language.
 *
 * Rules: (a) both `,` and `.` present: the right-most one is the decimal mark, the other the
 * thousands mark; (b) one kind, repeated: thousands marks (groups of exactly three digits);
 * (c) one occurrence followed by 1-2 digits (or none): decimal mark; (d) one occurrence followed
 * by exactly three digits with a 1-3 digit non-zero integer part: thousands mark (`1,234` is
 * 1234, `0,125` is 0.125); (e) otherwise decimal.
 */
export function parseAmountInput(text: string): number | null {
  const s = text.replace(/[\s\u00a0\u202f]/g, '')
  if (!/^[0-9.,]+$/.test(s) || !/[0-9]/.test(s)) return null

  const lastComma = s.lastIndexOf(',')
  const lastDot = s.lastIndexOf('.')
  let integer: string
  let fraction = ''

  if (lastComma !== -1 && lastDot !== -1) {
    const decimalAt = Math.max(lastComma, lastDot)
    const decimalMark = s[decimalAt]
    const thousandsMark = decimalMark === ',' ? '.' : ','
    const before = s.slice(0, decimalAt)
    fraction = s.slice(decimalAt + 1)
    if (before.includes(decimalMark) || fraction.includes(thousandsMark)) return null
    const groups = before.split(thousandsMark)
    if (!groups.every((g, i) => (i === 0 ? /^\d{1,3}$/.test(g) : /^\d{3}$/.test(g)))) return null
    integer = groups.join('')
  } else if (lastComma !== -1 || lastDot !== -1) {
    const mark = lastComma !== -1 ? ',' : '.'
    const parts = s.split(mark)
    if (parts.length > 2) {
      if (!/^\d{1,3}$/.test(parts[0]) || !parts.slice(1).every((p) => /^\d{3}$/.test(p))) return null
      integer = parts.join('')
    } else {
      const [head, tail] = parts
      if (tail.length === 3 && /^[1-9]\d{0,2}$/.test(head)) integer = head + tail
      else {
        integer = head
        fraction = tail
      }
    }
  } else {
    integer = s
  }

  if (!/^\d*$/.test(integer) || !/^\d*$/.test(fraction)) return null
  const value = Number(`${integer || '0'}${fraction ? `.${fraction}` : ''}`)
  return Number.isFinite(value) ? value : null
}
```

```ts
// Polish public holidays and business days (S03). Dates are calendar days as Date at 00:00 UTC;
// only UTC getters are used. Used by NBP rates (S04), schedules (S11) and the cash gap forecast (S13).

const DAY_MS = 24 * 60 * 60 * 1000

const utc = (year: number, month: number, day: number) => new Date(Date.UTC(year, month - 1, day))
const addDays = (date: Date, days: number) => new Date(date.getTime() + days * DAY_MS)
const dayStart = (date: Date) => utc(date.getUTCFullYear(), date.getUTCMonth() + 1, date.getUTCDate())

/** Easter Sunday (Meeus/Jones/Butcher, Gregorian). */
export function easterSunday(year: number): Date {
  const a = year % 19
  const b = Math.floor(year / 100)
  const c = year % 100
  const d = Math.floor(b / 4)
  const e = b % 4
  const f = Math.floor((b + 8) / 25)
  const g = Math.floor((b - f + 1) / 3)
  const h = (19 * a + b - d - g + 15) % 30
  const i = Math.floor(c / 4)
  const k = c % 4
  const l = (32 + 2 * e + 2 * i - h - k) % 7
  const m = Math.floor((a + 11 * h + 22 * l) / 451)
  const month = Math.floor((h + l - 7 * m + 114) / 31)
  const day = ((h + l - 7 * m + 114) % 31) + 1
  return utc(year, month, day)
}

/**
 * Public holidays (dni wolne od pracy) of a year, ascending. Epiphany (6 Jan) since 2011, Christmas
 * Eve (24 Dec) since 2025. Easter Sunday, Pentecost Sunday and the other Sundays are listed too.
 */
export function polishHolidays(year: number): Date[] {
  const easter = easterSunday(year)
  const holidays = [
    utc(year, 1, 1),
    ...(year >= 2011 ? [utc(year, 1, 6)] : []),
    easter,
    addDays(easter, 1),
    utc(year, 5, 1),
    utc(year, 5, 3),
    addDays(easter, 49), // Pentecost Sunday
    addDays(easter, 60), // Corpus Christi
    utc(year, 8, 15),
    utc(year, 11, 1),
    utc(year, 11, 11),
    ...(year >= 2025 ? [utc(year, 12, 24)] : []),
    utc(year, 12, 25),
    utc(year, 12, 26),
  ]
  return holidays.sort((x, y) => x.getTime() - y.getTime())
}

const holidayCache = new Map<number, Set<number>>()

function holidaySet(year: number): Set<number> {
  let set = holidayCache.get(year)
  if (!set) {
    set = new Set(polishHolidays(year).map((d) => d.getTime()))
    holidayCache.set(year, set)
  }
  return set
}

/** Monday to Friday that is not a public holiday. The time of day is ignored. */
export function isBusinessDay(date: Date): boolean {
  const day = dayStart(date)
  const weekday = day.getUTCDay()
  if (weekday === 0 || weekday === 6) return false
  return !holidaySet(day.getUTCFullYear()).has(day.getTime())
}

/** The nearest business day strictly before `date`. */
export function previousBusinessDay(date: Date): Date {
  let day = addDays(dayStart(date), -1)
  while (!isBusinessDay(day)) day = addDays(day, -1)
  return day
}

/** The nearest business day strictly after `date`. */
export function nextBusinessDay(date: Date): Date {
  let day = addDays(dayStart(date), 1)
  while (!isBusinessDay(day)) day = addDays(day, 1)
  return day
}

/** Moves `count` business days (negative = back). `count` 0 returns the day itself. */
export function addBusinessDays(date: Date, count: number): Date {
  let day = dayStart(date)
  const step = count < 0 ? -1 : 1
  for (let left = Math.abs(count); left > 0; left--) {
    day = step === 1 ? nextBusinessDay(day) : previousBusinessDay(day)
  }
  return day
}
```

```ts
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { join, relative, sep } from 'node:path'
import { describe, expect, it } from 'vitest'

const ROOT = join(import.meta.dirname, '..', '..', '..')

// Directories scanned. The marketing pages (app/page.tsx, app/pricing, components/landing) are
// out of S03's scope (S17), so they are not scanned.
const SCAN = ['app/(app)', 'app/(auth)', 'components', 'lib', 'types']
const SKIP_PREFIXES = ['components/landing']
// File -> why it may keep Cyrillic or 'uk-UA'.
const ALLOW: Record<string, string> = {
  'lib/i18n/config.ts': 'the Ukrainian locale tag and its own-language label',
}

const SOURCE = /\.(ts|tsx)$/

function walk(dir: string): string[] {
  const out: string[] = []
  for (const entry of readdirSync(dir)) {
    const full = join(dir, entry)
    if (statSync(full).isDirectory()) out.push(...walk(full))
    else if (SOURCE.test(entry)) out.push(full)
  }
  return out
}

// Removes block and line comments (not the // of a URL) so explanations in Ukrainian are allowed.
function stripComments(source: string): string {
  return source.replace(/\/\*[\s\S]*?\*\//g, '').replace(/(?<!:)\/\/.*$/gm, '')
}

const files = SCAN.flatMap((dir) => walk(join(ROOT, dir)))
  .map((file) => relative(ROOT, file).split(sep).join('/'))
  .filter((file) => !SKIP_PREFIXES.some((p) => file.startsWith(p)))

describe('no hard-coded UI text', () => {
  it('scans a sensible number of files', () => {
    expect(files.length).toBeGreaterThan(50)
  })

  it('no Cyrillic text outside the message catalogs', () => {
    const offenders = files.filter((file) => {
      if (ALLOW[file]) return false
      return /[А-Яа-яІіЇїЄєҐґ]/.test(stripComments(readFileSync(join(ROOT, file), 'utf8')))
    })
    expect(offenders).toEqual([])
  })

  it("no hard-coded 'uk-UA' / toLocaleDateString() / toLocaleString() in app code", () => {
    const offenders = files.filter((file) => {
      if (ALLOW[file]) return false
      return /uk-UA|\.toLocaleDateString\(|\.toLocaleString\(|\.toLocaleTimeString\(/.test(stripComments(readFileSync(join(ROOT, file), 'utf8')))
    })
    expect(offenders).toEqual([])
  })
})
```
