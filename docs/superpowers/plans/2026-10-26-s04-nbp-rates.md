# S04 NBP rates Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A server-side `resolveRate(from, to, date)` that returns a reproducible NBP table A rate snapshot (table strictly before the date), cached per month in Firestore, plus `GET /api/rates`.

**Architecture:** Pure helpers (`pickTable`, `crossRate`, day arithmetic) with no I/O; an NBP HTTP client; a Firestore month-document store behind a small interface; a `createResolver(deps)` that composes them with injected `store`, `fetchTables`, `now`, `today` (so every behaviour is unit-tested without network or Firestore); `index.ts` wires the real dependencies. The route validates params and maps `FxError` codes to HTTP statuses.

**Tech Stack:** TypeScript, Next.js 16 route handler, firebase-admin (Admin SDK, server only), Vitest, `@firebase/rules-unit-testing`. No new dependencies.

Spec: `docs/superpowers/specs/2026-10-26-s04-nbp-rates-design.md`. Work on branch `s04-nbp-rates` (already created, spec committed).

```ts
/** One published NBP table A: PLN per one unit of each currency (`mid`). */
export type NbpTable = {
  /** e.g. "208/A/NBP/2026" */
  no: string
  /** Publication day, YYYY-MM-DD. */
  effectiveDate: string
  rates: Record<string, number>
}

/** Cached `fxTables/{YYYY-MM}` document. */
export type MonthDoc = {
  month: string
  /** Ascending by effectiveDate. */
  tables: NbpTable[]
  /** Epoch ms of the NBP fetch. */
  fetchedAt: number
  /** True when the month had ended before the fetch: the document never changes again. */
  complete: boolean
}

/** What a caller stores next to an amount so the conversion can be reproduced. */
export type RateSnapshot = {
  rate: number
  from: string
  to: string
  requestedDate: string
  /** effectiveDate of the table used; equals requestedDate for identical currencies. */
  rateDate: string
  /** null for identical currencies. */
  tableNo: string | null
  source: 'NBP_A' | 'identity'
}

export type FxErrorCode = 'invalid_params' | 'unsupported_currency' | 'no_rate_before' | 'provider_unavailable'

export class FxError extends Error {
  public readonly code: FxErrorCode

  constructor(code: FxErrorCode) {
    super(code)
    this.name = 'FxError'
    this.code = code
  }
}
```

```ts
const DAY = /^\d{4}-\d{2}-\d{2}$/

/** True for a real calendar day written as YYYY-MM-DD. */
export function isDay(s: unknown): s is string {
  if (typeof s !== 'string' || !DAY.test(s)) return false
  const d = new Date(`${s}T00:00:00Z`)
  return !Number.isNaN(d.getTime()) && d.toISOString().slice(0, 10) === s
}

export function addDays(day: string, n: number): string {
  const d = new Date(`${day}T00:00:00Z`)
  d.setUTCDate(d.getUTCDate() + n)
  return d.toISOString().slice(0, 10)
}

/** "2026-10-26" -> "2026-10" */
export function monthOf(day: string): string {
  return day.slice(0, 7)
}

export function prevMonth(month: string): string {
  const [y, m] = month.split('-').map(Number)
  return new Date(Date.UTC(y, m - 2, 1)).toISOString().slice(0, 7)
}

export function monthBounds(month: string): { first: string; last: string } {
  const [y, m] = month.split('-').map(Number)
  return { first: `${month}-01`, last: new Date(Date.UTC(y, m, 0)).toISOString().slice(0, 10) }
}
```

```ts
import { FxError, type NbpTable } from '@/lib/server/fx/types'

/** Rates are stored and returned with 8 decimals (as Finmap shows them). */
export function round8(x: number): number {
  return Number(x.toFixed(8))
}

/**
 * The table with the greatest effectiveDate strictly before `day` (NBP rule: the last working day
 * BEFORE the operation). `tables` must be ascending. Weekends and holidays need no calendar: no
 * table is published on them.
 */
export function pickTable(tables: readonly NbpTable[], day: string): NbpTable | null {
  for (let i = tables.length - 1; i >= 0; i--) {
    if (tables[i].effectiveDate < day) return tables[i]
  }
  return null
}

function mid(table: NbpTable, code: string): number {
  if (code === 'PLN') return 1
  if (!Object.hasOwn(table.rates, code)) throw new FxError('unsupported_currency')
  return table.rates[code]
}

/** Unrounded rate converting one unit of `from` into `to` through PLN. */
export function crossRate(table: NbpTable, from: string, to: string): number {
  return mid(table, from) / mid(table, to)
}
```

```ts
import { isDay } from '@/lib/server/fx/days'
import { FxError, type NbpTable } from '@/lib/server/fx/types'

const BASE = 'https://api.nbp.pl/api/exchangerates/tables/A'
const TIMEOUT_MS = 8000

type FetchLike = (url: string, init?: { signal?: AbortSignal; cache?: RequestCache }) => Promise<Response>

function parseTables(json: unknown): NbpTable[] {
  if (!Array.isArray(json)) throw new FxError('provider_unavailable')
  const tables = json.map((raw): NbpTable => {
    if (typeof raw !== 'object' || raw === null) throw new FxError('provider_unavailable')
    const r = raw as Record<string, unknown>
    if (typeof r.no !== 'string' || !isDay(r.effectiveDate) || !Array.isArray(r.rates)) {
      throw new FxError('provider_unavailable')
    }
    const rates: Record<string, number> = {}
    for (const item of r.rates) {
      const it = (typeof item === 'object' && item !== null ? item : {}) as Record<string, unknown>
      if (typeof it.code !== 'string' || !/^[A-Z]{3}$/.test(it.code)) throw new FxError('provider_unavailable')
      if (typeof it.mid !== 'number' || !Number.isFinite(it.mid) || it.mid <= 0) throw new FxError('provider_unavailable')
      rates[it.code] = it.mid
    }
    return { no: r.no, effectiveDate: r.effectiveDate, rates }
  })
  return tables.sort((a, b) => (a.effectiveDate < b.effectiveDate ? -1 : 1))
}

/**
 * NBP table A for [fromDay, toDay] (at most 93 days, not in the future). 404 means no table was
 * published in the range. Every other failure is `provider_unavailable`; the NBP text never
 * reaches the caller.
 */
export async function fetchNbpTables(
  fromDay: string,
  toDay: string,
  fetchImpl: FetchLike = fetch,
): Promise<NbpTable[]> {
  let res: Response
  try {
    res = await fetchImpl(`${BASE}/${fromDay}/${toDay}/?format=json`, {
      signal: AbortSignal.timeout(TIMEOUT_MS),
      cache: 'no-store',
    })
  } catch {
    throw new FxError('provider_unavailable')
  }
  if (res.status === 404) return []
  if (!res.ok) throw new FxError('provider_unavailable')
  let json: unknown
  try {
    json = await res.json()
  } catch {
    throw new FxError('provider_unavailable')
  }
  return parseTables(json)
}
```

```ts
import 'server-only'
import { isDay } from '@/lib/server/fx/days'
import type { MonthDoc, NbpTable } from '@/lib/server/fx/types'

export interface FxStore {
  get(month: string): Promise<MonthDoc | null>
  set(doc: MonthDoc): Promise<void>
}

// The slice of the Admin Firestore this store needs (keeps it testable without the SDK).
type DbLike = {
  collection(name: string): {
    doc(id: string): {
      get(): Promise<{ exists: boolean; data(): unknown }>
      set(value: object): Promise<unknown>
    }
  }
}

const COLLECTION = 'fxTables'

function isTable(raw: unknown): raw is NbpTable {
  if (typeof raw !== 'object' || raw === null) return false
  const t = raw as Record<string, unknown>
  if (typeof t.no !== 'string' || !isDay(t.effectiveDate)) return false
  if (typeof t.rates !== 'object' || t.rates === null) return false
  return Object.values(t.rates).every((v) => typeof v === 'number' && Number.isFinite(v) && v > 0)
}

/** A stored document that does not look right is a cache miss, never an error. */
export function parseMonthDoc(raw: unknown, month: string): MonthDoc | null {
  if (typeof raw !== 'object' || raw === null) return null
  const d = raw as Record<string, unknown>
  if (d.month !== month || !Array.isArray(d.tables) || !d.tables.every(isTable)) return null
  if (typeof d.fetchedAt !== 'number' || !Number.isFinite(d.fetchedAt)) return null
  if (typeof d.complete !== 'boolean') return null
  return { month, tables: d.tables, fetchedAt: d.fetchedAt, complete: d.complete }
}

export function firestoreFxStore(db: DbLike): FxStore {
  return {
    async get(month) {
      const snap = await db.collection(COLLECTION).doc(month).get()
      return snap.exists ? parseMonthDoc(snap.data(), month) : null
    },
    async set(doc) {
      await db.collection(COLLECTION).doc(doc.month).set(doc)
    },
  }
}
```

```ts
/** Behaves like the NBP API: a future or inverted range or one over 93 days is an error. */
export function fakeNbp(tables: NbpTable[], today: string) {
  const calls: Array<[string, string]> = []
  const fn = async (from: string, to: string): Promise<NbpTable[]> => {
    calls.push([from, to])
    if (from > to || to > today || addDays(from, 93) < to) throw new FxError('provider_unavailable')
    return tables.filter((t) => t.effectiveDate >= from && t.effectiveDate <= to)
  }
  return { fn, calls }
}

      sets.push(doc.month)
```

```ts
import { addDays, isDay, monthBounds, monthOf, prevMonth } from '@/lib/server/fx/days'
import { crossRate, pickTable, round8 } from '@/lib/server/fx/rates'
import type { FxStore } from '@/lib/server/fx/store'
import { FxError, type MonthDoc, type NbpTable, type RateSnapshot } from '@/lib/server/fx/types'

const HOUR_MS = 3_600_000
const CODE = /^[A-Z]{3}$/
// First month with published tables (2002-01-02).
const FIRST_MONTH = '2002-01'

export type FxDeps = {
  store: FxStore
  fetchTables: (fromDay: string, toDay: string) => Promise<NbpTable[]>
  /** Epoch ms. */
  now: () => number
  /** Warsaw calendar day, YYYY-MM-DD. */
  today: () => string
}

export function createResolver(deps: FxDeps) {
  async function loadMonth(month: string, day: string): Promise<MonthDoc> {
    const today = deps.today()
    const { first, last } = monthBounds(month)
    // NBP answers 400 for a range that reaches into the future: such a month has nothing yet.
    if (first > today) return { month, tables: [], fetchedAt: deps.now(), complete: false }

    const cached = await deps.store.get(month)
    if (cached && (cached.complete || deps.now() - cached.fetchedAt < HOUR_MS)) return cached

    const finished = last < today
    try {
      const tables = await deps.fetchTables(first, finished ? last : today)
      // A finished month always has tables; empty means NBP misbehaved, so do not cache it forever.
      if (finished && tables.length === 0) throw new FxError('provider_unavailable')
      const doc: MonthDoc = { month, tables, fetchedAt: deps.now(), complete: finished }
      try {
        await deps.store.set(doc)
      } catch (e) {
        console.error('fx cache write failed', { name: (e as Error)?.name })
      }
      return doc
    } catch (e) {
      // A stale month is safe to serve only if it already holds a table on or after `day`.
      if (cached && cached.tables.some((t) => t.effectiveDate >= day)) return cached
      throw e instanceof FxError ? e : new FxError('provider_unavailable')
    }
  }

  return async function resolveRate(from: string, to: string, date: string): Promise<RateSnapshot> {
    if (!CODE.test(from) || !CODE.test(to) || !isDay(date)) throw new FxError('invalid_params')
    if (date > addDays(deps.today(), 366)) throw new FxError('invalid_params')

    if (from === to) {
      return { rate: 1, from, to, requestedDate: date, rateDate: date, tableNo: null, source: 'identity' }
    }

    let month = monthOf(addDays(date, -1))
    while (month >= FIRST_MONTH) {
      const doc = await loadMonth(month, date)
      const table = pickTable(doc.tables, date)
      if (table) {
        return {
          rate: round8(crossRate(table, from, to)),
          from,
          to,
          requestedDate: date,
          rateDate: table.effectiveDate,
          tableNo: table.no,
          source: 'NBP_A',
        }
      }
      month = prevMonth(month)
    }
    throw new FxError('no_rate_before')
  }
}
```

```ts
export function resolveRate(from: string, to: string, date: string): Promise<RateSnapshot> {

    store: firestoreFxStore(getAdmin().db),
    fetchTables: (a, b) => fetchNbpTables(a, b),
    now: () => Date.now(),
    today: () => todayUtcDay().toISOString().slice(0, 10),
  })(from, to, date)
}
```

```ts
import { requireUser } from '@/lib/server/auth'
import { FxError, resolveRate } from '@/lib/server/fx'
import { errorResponse, HttpError } from '@/lib/server/http-error'

export const runtime = 'nodejs'

const PARAMS = ['from', 'to', 'date']
const STATUS = {
  invalid_params: 400,
  unsupported_currency: 404,
  no_rate_before: 404,
  provider_unavailable: 503,
} as const

// GET /api/rates?from=EUR&to=PLN&date=2026-10-27
// NBP table A rate for converting `from` into `to` on `date` (S04). Public data, so any signed-in
// user may ask; no company is involved.
export async function GET(req: Request) {
  try {
    await requireUser(req)

    const params = new URL(req.url).searchParams
    const keys = [...params.keys()]
    if (keys.length !== PARAMS.length || !PARAMS.every((k) => params.getAll(k).length === 1)) {
      throw new HttpError(400, 'invalid_params')
    }

    const snapshot = await resolveRate(params.get('from')!, params.get('to')!, params.get('date')!)
    return Response.json(snapshot, { headers: { 'Cache-Control': 'private, max-age=300' } })
  } catch (e) {
    if (e instanceof FxError) return errorResponse(new HttpError(STATUS[e.code], e.code))
    return errorResponse(e)
  }
}
```
