**Goal:** A journal that reads operations page by page (up to 10 000), shows the account balance after every operation, filters by two dates, changes and deletes selected rows in bulk, and exports CSV for the accountant.

**Architecture:** Pure domain in `lib/domain/journal/` (order, filters, balance, CSV, bulk selection rules). A server module `lib/server/journal/` answers `GET /api/orgs/{orgId}/journal` (Admin SDK, a scan loop for in-memory filters, balance from daily rollups plus the same-day operations) and `POST .../transactions/bulk` (runs the existing `updateTransaction`/`deleteTransaction` one id at a time). The browser pages through the endpoint (hook `useJournal`), keeps filters in the URL, and builds the CSV itself by paging the same endpoint. Spec: `docs/superpowers/specs/2026-10-28-s08-journal-design.md`.

```ts
// lib/domain/journal/order.ts
import { isDay } from '@/lib/domain/ledger/days'
import { isId } from '@/lib/domain/ledger/guards'

/**
 * The listing order of the journal: payment date, then creation time (down to the nanosecond, as Firestore
 * stores it), then id; all descending. It is a total order, so a page boundary never splits ties.
 */
export interface JournalKey {
  paymentDate: string
  createdAtS: number
  createdAtN: number
  id: string
}

/** Negative when `a` is listed before `b`. */
export function compareKeys(a: JournalKey, b: JournalKey): number {
  if (a.paymentDate !== b.paymentDate) return a.paymentDate > b.paymentDate ? -1 : 1
  if (a.createdAtS !== b.createdAtS) return a.createdAtS > b.createdAtS ? -1 : 1
  if (a.createdAtN !== b.createdAtN) return a.createdAtN > b.createdAtN ? -1 : 1
  if (a.id !== b.id) return a.id > b.id ? -1 : 1
  return 0
}

/** Opaque to callers: URL-safe base64 of a JSON array. */
export function encodeCursor(k: JournalKey): string {
  const json = JSON.stringify([k.paymentDate, k.createdAtS, k.createdAtN, k.id])
  return btoa(json).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '')
}

export function decodeCursor(cursor: string): JournalKey | null {
  try {
    const padded = cursor.replace(/-/g, '+').replace(/_/g, '/')
    const parsed: unknown = JSON.parse(atob(padded))
    if (!Array.isArray(parsed) || parsed.length !== 4) return null
    const [paymentDate, s, n, id] = parsed as unknown[]
    if (!isDay(paymentDate) || !isId(id)) return null
    if (typeof s !== 'number' || !Number.isInteger(s) || s < 0) return null
    if (typeof n !== 'number' || !Number.isInteger(n) || n < 0 || n > 999_999_999) return null
    return { paymentDate, createdAtS: s, createdAtN: n, id }
  } catch {
    return null
  }
}
```

Note: the test builds a bad cursor with `btoa('[]')` (standard base64, padded); `decodeCursor` must accept standard and URL-safe forms, which the replace calls do (atob accepts missing padding in Node 22; if a runtime rejects it, re-add `=` padding: `padded + '='.repeat((4 - (padded.length % 4)) % 4)`; do that to be safe).

// lib/domain/journal/filter.ts
import { isDay } from '@/lib/domain/ledger/days'
import { isId } from '@/lib/domain/ledger/guards'
import { isMinor } from '@/lib/domain/ledger/money'
import { TX_STATUSES, TX_TYPES, type LedgerTransaction, type TxStatus, type TxType } from '@/lib/domain/ledger/types'

export type DateBasis = 'payment' | 'accrual'

export interface JournalFilters {
  /** Inclusive days, compared with the payment date or with the accrual date or period (`basis`). */
  from?: string
  to?: string
  basis: DateBasis
  /** Free text: account name, counterparty name, comment. */
  q?: string
  /** Inclusive, in the minor units of the operation's own currency. */
  minMinor?: number
  maxMinor?: number
  counterpartyIds: string[]
  types: TxType[]
  status?: TxStatus
  accountId?: string
  categoryIds: string[]
}

export const EMPTY_FILTERS: JournalFilters = { basis: 'payment', counterpartyIds: [], types: [], categoryIds: [] }

const MAX_LIST = 30
const MAX_QUERY = 100

/** Lower case, no diacritics; `ł` has no decomposition, so it is mapped by hand. */
export function normalizeText(s: string): string {
  return s
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .replace(/ł/gi, 'l')
    .toLowerCase()
}

function ids(raw: string | null): string[] {
  if (!raw) return []
  return raw
    .split(',')
    .filter((s) => isId(s))
    .slice(0, MAX_LIST)
}

function minor(raw: string | null): number | undefined {
  if (raw === null || !/^\d+$/.test(raw)) return undefined
  const n = Number(raw)
  return isMinor(n) ? n : undefined
}

/** Lenient on purpose: a hand-edited link with a bad value just loses that value. */
export function parseFilters(p: URLSearchParams): JournalFilters {
  const f: JournalFilters = { ...EMPTY_FILTERS, counterpartyIds: ids(p.get('cp')), categoryIds: ids(p.get('cat')) }
  const from = p.get('from')
  const to = p.get('to')
  if (isDay(from)) f.from = from
  if (isDay(to)) f.to = to
  if (p.get('basis') === 'accrual') f.basis = 'accrual'
  const q = p.get('q')?.trim()
  if (q) f.q = q.slice(0, MAX_QUERY)
  const min = minor(p.get('min'))
  const max = minor(p.get('max'))
  if (min !== undefined) f.minMinor = min
  if (max !== undefined) f.maxMinor = max
  f.types = (p.get('types') ?? '')
    .split(',')
    .filter((t): t is TxType => (TX_TYPES as readonly string[]).includes(t))
  const status = p.get('status')
  if (status !== null && (TX_STATUSES as readonly string[]).includes(status)) f.status = status as TxStatus
  const account = p.get('account')
  if (isId(account)) f.accountId = account
  return f
}

export function filtersToParams(f: JournalFilters): URLSearchParams {
  const p = new URLSearchParams()
  if (f.from) p.set('from', f.from)
  if (f.to) p.set('to', f.to)
  if (f.basis === 'accrual') p.set('basis', 'accrual')
  if (f.q) p.set('q', f.q)
  if (f.minMinor !== undefined) p.set('min', String(f.minMinor))
  if (f.maxMinor !== undefined) p.set('max', String(f.maxMinor))
  if (f.counterpartyIds.length) p.set('cp', f.counterpartyIds.join(','))
  if (f.types.length) p.set('types', f.types.join(','))
  if (f.status) p.set('status', f.status)
  if (f.accountId) p.set('account', f.accountId)
  if (f.categoryIds.length) p.set('cat', f.categoryIds.join(','))
  return p
}

/** True when something narrows the list; the date basis alone narrows nothing. */
export function hasActiveFilters(f: JournalFilters): boolean {
  return (
    f.from !== undefined ||
    f.to !== undefined ||
    !!f.q ||
    f.minMinor !== undefined ||
    f.maxMinor !== undefined ||
    f.counterpartyIds.length > 0 ||
    f.types.length > 0 ||
    f.status !== undefined ||
    f.accountId !== undefined ||
    f.categoryIds.length > 0
  )
}

/** Names the text search looks in. */
export interface FilterContext {
  accountName: (id: string) => string | undefined
  counterpartyName: (id: string) => string | undefined
}

function accrualBounds(tx: LedgerTransaction): { from: string; to: string } {
  return tx.accrual.mode === 'date' ? { from: tx.accrual.date, to: tx.accrual.date } : { from: tx.accrual.from, to: tx.accrual.to }
}

/** The whole predicate, so the in-memory part of a page read and the tests use one definition. */
export function matchesFilters(tx: LedgerTransaction, f: JournalFilters, ctx: FilterContext): boolean {
  if (f.types.length && !f.types.includes(tx.type)) return false
  if (f.status && tx.status !== f.status) return false
  if (f.minMinor !== undefined && tx.amountMinor < f.minMinor) return false
  if (f.maxMinor !== undefined && tx.amountMinor > f.maxMinor) return false
  if (f.categoryIds.length && (tx.categoryId === undefined || !f.categoryIds.includes(tx.categoryId))) return false
  if (f.counterpartyIds.length && (tx.counterpartyId === undefined || !f.counterpartyIds.includes(tx.counterpartyId))) return false
  if (f.accountId && tx.accountId !== f.accountId && tx.toAccountId !== f.accountId) return false

  if (f.from !== undefined || f.to !== undefined) {
    const range = f.basis === 'payment' ? { from: tx.paymentDate, to: tx.paymentDate } : accrualBounds(tx)
    if (f.from !== undefined && range.to < f.from) return false
    if (f.to !== undefined && range.from > f.to) return false
  }

  if (f.q) {
    const hay = [
      ctx.accountName(tx.accountId),
      tx.toAccountId === undefined ? undefined : ctx.accountName(tx.toAccountId),
      tx.counterpartyId === undefined ? undefined : ctx.counterpartyName(tx.counterpartyId),
      tx.description,
    ]
      .filter((s): s is string => !!s)
      .map(normalizeText)
    const words = normalizeText(f.q).split(/\s+/).filter(Boolean)
    if (!words.every((w) => hay.some((h) => h.includes(w)))) return false
  }
  return true
}
```

// lib/domain/journal/balance.ts
export interface DayOps {
  day: string
  /** Every actual operation of the account on that day, in listing order (newest first), with its effect on the account. */
  ops: { id: string; effect: number }[]
}

/**
 * Balance of one account after each operation of the days on a page.
 *
 * `days` are in listing order (newest first); `netAfterFirstDay` is the net movement of the account on all days newer
 * than `days[0]`; `netBetween[i]` is the net movement on the days strictly between `days[i]` and `days[i + 1]`.
 * Walking from the balance now: the balance after the newest operation of a day is the balance at the end of that day;
 * removing each operation's effect gives the balance after the one listed next.
 */
export function balancesAfter(balanceNow: number, netAfterFirstDay: number, days: DayOps[], netBetween: number[]): Map<string, number> {
  const out = new Map<string, number>()
  let running = balanceNow - netAfterFirstDay
  days.forEach((d, i) => {
    for (const op of d.ops) {
      out.set(op.id, running)
      running -= op.effect
    }
    running -= netBetween[i] ?? 0
  })
  return out
}

export const EXPORT_COLUMNS = [
  'paymentDate',
  'accrualFrom',
  'accrualTo',
  'status',
  'type',
  'account',
  'toAccount',
  'category',
  'counterparty',
  'comment',
  'amount',
  'currency',
  'amountBase',
  'rate',
  'rateDate',
  'balanceAfter',
] as const

export type ExportColumn = (typeof EXPORT_COLUMNS)[number]

export interface ExportContext {
  /** The decimal sign of the language: ',' for pl and uk, '.' for en. */
  decimal: ',' | '.'
  accountName: (id: string) => string
  categoryName: (id: string) => string
  counterpartyName: (id: string) => string
  statusLabel: (status: string) => string
  typeLabel: (type: string) => string
  baseCurrency: string
}

const DELIMITER = ';'

/** Minor units with two decimals, the language's decimal sign and no thousands separator. */
export function formatAmount(minor: number, decimal: ',' | '.'): string {
  const sign = minor < 0 ? '-' : ''
  const abs = Math.abs(minor)
  const whole = Math.trunc(abs / 100)
  const frac = String(abs % 100).padStart(2, '0')
  return `${sign}${whole}${decimal}${frac}`
}

/** Quotes where needed and defuses spreadsheet formulas (a cell that starts with = + - @ or a tab). */
export function csvCell(value: string): string {
  const safe = /^[=+\-@\t\r]/.test(value) ? `'${value}` : value
  return /[;"\r\n]/.test(safe) ? `"${safe.replace(/"/g, '""')}"` : safe
}

/** UTF-8 BOM so Excel reads the diacritics, `;` as the delimiter, CRLF line ends. */
export function buildCsv(header: string[], rows: string[][]): string {
  const lines = [header, ...rows].map((cells) => cells.map(csvCell).join(DELIMITER))
  return `﻿${lines.join('\r\n')}\r\n`
}

export function exportRow({ tx, balanceAfterMinor }: JournalRow, ctx: ExportContext): string[] {
  const accrual = tx.accrual.mode === 'date' ? { from: tx.accrual.date, to: tx.accrual.date } : { from: tx.accrual.from, to: tx.accrual.to }
  const sign = tx.type === 'income' ? 1 : -1
  const base = tx.type === 'transfer' ? null : baseOf(tx, ctx.baseCurrency)
  const evidence = tx.fx?.payment
  const cells: Record<ExportColumn, string> = {
    paymentDate: tx.paymentDate,
    accrualFrom: tx.type === 'transfer' ? '' : accrual.from,
    accrualTo: tx.type === 'transfer' ? '' : accrual.to,
    status: ctx.statusLabel(tx.status),
    type: ctx.typeLabel(tx.type),
    account: ctx.accountName(tx.accountId),
    toAccount: tx.toAccountId === undefined ? '' : ctx.accountName(tx.toAccountId),
    category: tx.categoryId === undefined ? '' : ctx.categoryName(tx.categoryId),
    counterparty: tx.counterpartyId === undefined ? '' : ctx.counterpartyName(tx.counterpartyId),
    comment: tx.description ?? '',
    amount: formatAmount(sign * tx.amountMinor, ctx.decimal),
    currency: tx.currency,
    amountBase: base ? formatAmount(sign * base.payment, ctx.decimal) : '',
    rate: evidence ? String(evidence.rate).replace('.', ctx.decimal) : '',
    rateDate: evidence?.rateDate ?? '',
    balanceAfter: balanceAfterMinor === null ? '' : formatAmount(balanceAfterMinor, ctx.decimal),
  }
  return EXPORT_COLUMNS.map((c) => cells[c])
}

// lib/domain/journal/selection.ts
import { baseOf } from '@/lib/domain/ledger/build'
import type { Accrual, LedgerTransaction, TxType } from '@/lib/domain/ledger/types'

/** What a bulk change may set. `null` clears the field. */
export interface BulkPatch {
  categoryId?: string | null
  counterpartyId?: string | null
  description?: string | null
  paymentDate?: string
  accrual?: Accrual
}

export type SkipReason = 'transfer' | 'future_actual'

/** Categories have a type, so a category can be set only on a selection of one type; transfers have none. */
export function categoryTypeOf(rows: LedgerTransaction[]): 'income' | 'expense' | null {
  const types = new Set<TxType>(rows.filter((r) => r.type !== 'transfer').map((r) => r.type))
  if (types.size !== 1) return null
  return types.has('income') ? 'income' : 'expense'
}

/** A transfer has no category, counterparty or accrual (it has no P&L side). */
export function patchTouches(patch: BulkPatch, type: TxType): boolean {
  return type === 'transfer' && ('categoryId' in patch || 'counterpartyId' in patch || 'accrual' in patch)
}

/** Splits a selection into what the patch applies to and what is skipped, before anything is sent. */
export function splitByPatch(
  rows: LedgerTransaction[],
  patch: BulkPatch,
  today: string,
): { apply: LedgerTransaction[]; skipped: { id: string; reason: SkipReason }[] } {
  const apply: LedgerTransaction[] = []
  const skipped: { id: string; reason: SkipReason }[] = []
  for (const tx of rows) {
    if (patchTouches(patch, tx.type)) skipped.push({ id: tx.id, reason: 'transfer' })
    else if (patch.paymentDate !== undefined && patch.paymentDate > today && tx.status === 'actual') skipped.push({ id: tx.id, reason: 'future_actual' })
    else apply.push(tx)
  }
  return { apply, skipped }
}

/** The number of rows and the net company-currency amount of the actual income and expense among them. */
export function selectionSummary(rows: LedgerTransaction[], baseCurrency: string): { count: number; totalBaseMinor: number } {
  let total = 0
  for (const tx of rows) {
    if (tx.status !== 'actual' || tx.type === 'transfer') continue
    const base = baseOf(tx, baseCurrency)
    if (!base) continue
    total += tx.type === 'income' ? base.payment : -base.payment
  }
  return { count: rows.length, totalBaseMinor: total }
}

// lib/domain/journal/bulk-input.ts
import { isDay } from '@/lib/domain/ledger/days'
import { asRecord, bad, isId, onlyKeys } from '@/lib/domain/ledger/guards'
import { parseTransactionPatch } from '@/lib/domain/ledger/input'
import type { BulkPatch } from './selection'

/** One request carries at most this many operations; the browser sends a larger selection in several requests. */
export const MAX_BULK_IDS = 25

export type BulkRequest = { action: 'delete'; ids: string[] } | { action: 'update'; ids: string[]; patch: BulkPatch }

const PATCH_KEYS = ['categoryId', 'counterpartyId', 'description', 'paymentDate', 'accrual'] as const

export function parseBulkRequest(body: unknown): BulkRequest {
  const r = asRecord(body)
  onlyKeys(r, ['action', 'ids', 'patch'])
  const ids = r.ids
  if (!Array.isArray(ids) || ids.length === 0 || ids.length > MAX_BULK_IDS) bad('ids')
  if (!ids.every((id) => isId(id)) || new Set(ids).size !== ids.length) bad('ids')

  if (r.action === 'delete') {
    if (r.patch !== undefined) bad('patch')
    return { action: 'delete', ids: ids as string[] }
  }
  if (r.action !== 'update') bad('action')

  const patch = asRecord(r.patch, 'patch')
  onlyKeys(patch, PATCH_KEYS, 'patch.')
  if (Object.keys(patch).length === 0) bad('patch')
  if (patch.paymentDate !== undefined && !isDay(patch.paymentDate)) bad('patch.paymentDate')
  // The field-level rules (types, nulls, accrual shape) are the ones the single-operation PATCH enforces.
  parseTransactionPatch(patch)
  return { action: 'update', ids: ids as string[], patch: patch as BulkPatch }
}

}

    cursor = page.next === null ? null : decodeCursor(page.next)
    if (++guard > 200) throw new Error('the cursor never ends')
  } while (cursor)
  return rows
}

// lib/server/journal/read.ts
import { AggregateField, FieldPath, Filter, Timestamp, type DocumentSnapshot, type Query } from 'firebase-admin/firestore'
import { balancesAfter, type DayOps } from '@/lib/domain/journal/balance'
import { matchesFilters, type FilterContext, type JournalFilters } from '@/lib/domain/journal/filter'
import { compareKeys, encodeCursor, type JournalKey } from '@/lib/domain/journal/order'
import type { JournalPage, JournalRow } from '@/lib/domain/journal/types'
import { parseAccount, parseTransaction } from '@/lib/domain/ledger/doc'
import { balanceEffects } from '@/lib/domain/ledger/effects'
import { cashDocId } from '@/lib/domain/ledger/rollups'
import type { LedgerAccount } from '@/lib/domain/ledger/types'
import type { LedgerDeps } from '@/lib/server/ledger/deps'
import { accountsCol, counterpartiesCol, rollupsCol, transactionsCol } from '@/lib/server/ledger/paths'

/** Documents one request may read while looking for a full page. */
export const SCAN_CAP = 2000
const SCAN_CHUNK = 200

export interface ReadOptions {
  filters: JournalFilters
  cursor: JournalKey | null
  limit: number
  /** For tests. */
  scanCap?: number
}

/** What a request read, for the budget test: operation documents, rollup documents, aggregate queries. */
export interface ReadStats {
  docsRead: number
  aggregates: number
}

export type ReadResult = JournalPage & { stats: ReadStats }

function keyOf(d: DocumentSnapshot): JournalKey | null {
  const created = d.get('createdAt')
  const paymentDate = d.get('paymentDate')
  if (!(created instanceof Timestamp) || typeof paymentDate !== 'string') return null
  return { paymentDate, createdAtS: created.seconds, createdAtN: created.nanoseconds, id: d.id }
}

function needsScan(f: JournalFilters): boolean {
  return (
    f.types.length > 0 ||
    f.status !== undefined ||
    f.minMinor !== undefined ||
    f.maxMinor !== undefined ||
    f.categoryIds.length > 0 ||
    f.counterpartyIds.length > 0 ||
    !!f.q ||
    (f.basis === 'accrual' && (f.from !== undefined || f.to !== undefined))
  )
}

/** The part of the filters Firestore answers through the indexes: the account and a payment-date range. */
function baseQuery(deps: LedgerDeps, orgId: string, f: JournalFilters): Query {
  let q: Query = transactionsCol(deps.db, orgId)
  if (f.accountId) q = q.where(Filter.or(Filter.where('accountId', '==', f.accountId), Filter.where('toAccountId', '==', f.accountId)))
  if (f.basis === 'payment') {
    if (f.from) q = q.where('paymentDate', '>=', f.from)
    if (f.to) q = q.where('paymentDate', '<=', f.to)
  }
  return q.orderBy('paymentDate', 'desc').orderBy('createdAt', 'desc').orderBy(FieldPath.documentId(), 'desc')
}

function after(q: Query, key: JournalKey): Query {
  return q.startAfter(key.paymentDate, new Timestamp(key.createdAtS, key.createdAtN), key.id)
}

export async function readJournalPage(deps: LedgerDeps, orgId: string, opts: ReadOptions): Promise<ReadResult> {
  const stats: ReadStats = { docsRead: 0, aggregates: 0 }
  const { filters, limit } = opts
  const cap = opts.scanCap ?? SCAN_CAP

  // Accounts (names, balances) and, for a text search, counterparty names: small collections, not counted as operations.
  const accounts = new Map<string, LedgerAccount>()
  for (const d of (await accountsCol(deps.db, orgId).get()).docs) {
    const a = parseAccount(d.id, d.data())
    if (a) accounts.set(a.id, a)
  }
  const counterpartyNames = new Map<string, string>()
  if (filters.q) {
    for (const d of (await counterpartiesCol(deps.db, orgId).get()).docs) {
      const name = d.get('name')
      if (typeof name === 'string') counterpartyNames.set(d.id, name)
    }
  }
  const ctx: FilterContext = { accountName: (id) => accounts.get(id)?.name, counterpartyName: (id) => counterpartyNames.get(id) }

  const base = baseQuery(deps, orgId, filters)
  const scan = needsScan(filters)
  const rows: JournalRow[] = []
  let lastKey = opts.cursor
  let scanned = 0
  let more = true
  let full = false

  while (rows.length < limit && scanned < cap && more) {
    const size = scan ? SCAN_CHUNK : Math.min(SCAN_CHUNK, limit - rows.length)
    const snap = await (lastKey ? after(base, lastKey) : base).limit(size).get()
    stats.docsRead += snap.size
    for (const d of snap.docs) {
      const key = keyOf(d)
      if (!key) continue
      lastKey = key
      scanned++
      const tx = parseTransaction(d.id, d.data())
      if (tx && matchesFilters(tx, filters, ctx)) {
        rows.push({ tx, balanceAfterMinor: null })
        if (rows.length === limit) {
          full = true
          break
        }
      }
    }
    if (snap.size < size) more = false
  }

  await attachBalances(deps, orgId, rows, accounts, stats)
  return { rows, next: lastKey !== null && (full || more) ? encodeCursor(lastKey) : null, stats }
}

// ---- balance after each operation ----------------------------------------------------------------------------

const END_OF_TIME = '9999-12-31'

interface DayNet {
  inMinor?: number
  outMinor?: number
  transferInMinor?: number
  transferOutMinor?: number
}

const netOf = (d: DayNet): number => (d.inMinor ?? 0) - (d.outMinor ?? 0) + (d.transferInMinor ?? 0) - (d.transferOutMinor ?? 0)

/** The net movement of an account on every day after `day` (one aggregate query over the daily rollup ids). */
async function netAfterDay(deps: LedgerDeps, orgId: string, accountId: string, day: string, stats: ReadStats): Promise<number> {
  const col = rollupsCol(deps.db, orgId)
  const agg = await col
    .where(FieldPath.documentId(), '>', col.doc(cashDocId(accountId, day)))
    .where(FieldPath.documentId(), '<=', col.doc(cashDocId(accountId, END_OF_TIME)))
    .aggregate({
      inMinor: AggregateField.sum('inMinor'),
      outMinor: AggregateField.sum('outMinor'),
      transferInMinor: AggregateField.sum('transferInMinor'),
      transferOutMinor: AggregateField.sum('transferOutMinor'),
    })
    .get()
  stats.aggregates++
  return netOf(agg.data())
}

/** Net movement per day for the days from `oldest` to `newest` that have a rollup. */
async function netsBetween(deps: LedgerDeps, orgId: string, accountId: string, oldest: string, newest: string, stats: ReadStats): Promise<Map<string, number>> {
  const col = rollupsCol(deps.db, orgId)
  const snap = await col
    .where(FieldPath.documentId(), '>=', col.doc(cashDocId(accountId, oldest)))
    .where(FieldPath.documentId(), '<=', col.doc(cashDocId(accountId, newest)))
    .get()
  stats.docsRead += snap.size
  const prefix = cashDocId(accountId, '')
  const out = new Map<string, number>()
  for (const d of snap.docs) out.set(d.id.slice(prefix.length), netOf(d.data() as DayNet))
  return out
}

/** Every actual operation that moves the account on `day`, in listing order, with its effect on the account. */
async function opsOfDay(deps: LedgerDeps, orgId: string, accountId: string, day: string, stats: ReadStats): Promise<DayOps['ops']> {
  const snap = await transactionsCol(deps.db, orgId)
    .where(Filter.and(Filter.where('paymentDate', '==', day), Filter.or(Filter.where('accountId', '==', accountId), Filter.where('toAccountId', '==', accountId))))
    .get()
  stats.docsRead += snap.size
  const listed: { key: JournalKey; id: string; effect: number }[] = []
  for (const d of snap.docs) {
    const tx = parseTransaction(d.id, d.data())
    if (!tx || tx.status !== 'actual') continue
    // An operation without a creation time cannot be listed, but it still moved the balance: keep it, last in its day.
    const key = keyOf(d) ?? { paymentDate: day, createdAtS: 0, createdAtN: 0, id: d.id }
    listed.push({ key, id: d.id, effect: balanceEffects(tx)[accountId] ?? 0 })
  }
  return listed.sort((a, b) => compareKeys(a.key, b.key)).map(({ id, effect }) => ({ id, effect }))
}

async function attachBalances(deps: LedgerDeps, orgId: string, rows: JournalRow[], accounts: Map<string, LedgerAccount>, stats: ReadStats): Promise<void> {
  const byAccount = new Map<string, JournalRow[]>()
  for (const row of rows) {
    if (row.tx.status !== 'actual') continue
    byAccount.set(row.tx.accountId, [...(byAccount.get(row.tx.accountId) ?? []), row])
  }

  for (const [accountId, accountRows] of byAccount) {
    const account = accounts.get(accountId)
    if (!account) continue
    const days = [...new Set(accountRows.map((r) => r.tx.paymentDate))].sort().reverse() // newest first
    const newest = days[0]
    const oldest = days[days.length - 1]

    const netAfter = await netAfterDay(deps, orgId, accountId, newest, stats)
    const nets = days.length > 1 ? await netsBetween(deps, orgId, accountId, oldest, newest, stats) : new Map<string, number>()
    const dayOps: DayOps[] = []
    for (const day of days) dayOps.push({ day, ops: await opsOfDay(deps, orgId, accountId, day, stats) })

    const netBetween = days.slice(0, -1).map((newer, i) => {
      const older = days[i + 1]
      let sum = 0
      for (const [day, net] of nets) if (day > older && day < newer) sum += net
      return sum
    })

    const result = balancesAfter(account.balanceMinor, netAfter, dayOps, netBetween)
    for (const row of accountRows) row.balanceAfterMinor = result.get(row.tx.id) ?? null
  }
}

// lib/server/journal/bulk.ts
import type { BulkRequest } from '@/lib/domain/journal/bulk-input'
import type { TransactionPatch } from '@/lib/domain/ledger/input'
import { HttpError } from '@/lib/server/http-error'
import { toHttpError } from '@/lib/server/ledger/http'

export type BulkOutcome = { id: string; ok: true } | { id: string; ok: false; code: string }

/** The two ledger calls a bulk change is made of (the production `ledger` object has them). */
export interface BulkLedger {
  updateTransaction(orgId: string, uid: string, txId: string, patch: TransactionPatch): Promise<unknown>
  deleteTransaction(orgId: string, uid: string, txId: string): Promise<unknown>
}

/**
 * One operation at a time through the same validation and Firestore transaction as an edit in the form, so balances,
 * rollups and the audit trail are right by construction. A refusal (a ledger or rate error) is that row's outcome and the
 * rest go on; anything else (the database is down) aborts the request.
 */
export async function runBulk(ledger: BulkLedger, orgId: string, uid: string, req: BulkRequest): Promise<BulkOutcome[]> {
  const out: BulkOutcome[] = []
  for (const id of req.ids) {
    try {
      if (req.action === 'delete') await ledger.deleteTransaction(orgId, uid, id)
      else await ledger.updateTransaction(orgId, uid, id, req.patch as TransactionPatch)
      out.push({ id, ok: true })
    } catch (e) {
      const mapped = toHttpError(e)
      if (mapped instanceof HttpError && (mapped.status < 500 || mapped.status === 503)) {
        out.push({ id, ok: false, code: mapped.message })
      } else {
        throw e
      }
    }
  }
  return out
}

import { MAX_BULK_IDS, type BulkOutcome, type BulkRequest } from '@/lib/domain/journal/bulk-input'
import { apiCall } from './ledger-client'
import { notifyLedgerChanged } from './ledger-events'

/** Sends one chunk (at most 25 ids) to the bulk endpoint. */
export async function postBulk(orgId: string, request: BulkRequest): Promise<BulkOutcome[]> {
  const out = (await apiCall('POST', `/api/orgs/${encodeURIComponent(orgId)}/transactions/bulk`, request)) as { results: BulkOutcome[] }
  return out.results
}

export interface ChunkedResult {
  outcomes: BulkOutcome[]
  /** True when the person stopped it: what was done stays done. */
  stopped: boolean
}

/** Runs `send` over the ids in chunks, one after another; the signal is checked between chunks. */
export async function runChunked(
  ids: string[],
  send: (chunk: string[]) => Promise<BulkOutcome[]>,
  opts: { signal?: AbortSignal; onProgress?: (done: number, total: number) => void } = {},
): Promise<ChunkedResult> {
  const outcomes: BulkOutcome[] = []
  for (let i = 0; i < ids.length; i += MAX_BULK_IDS) {
    if (opts.signal?.aborted) return { outcomes, stopped: true }
    outcomes.push(...(await send(ids.slice(i, i + MAX_BULK_IDS))))
    opts.onProgress?.(Math.min(i + MAX_BULK_IDS, ids.length), ids.length)
    if (opts.signal?.aborted && i + MAX_BULK_IDS < ids.length) return { outcomes, stopped: true }
  }
  return { outcomes, stopped: false }
}

/** Applies a bulk request to many ids, then tells the journal once that operations changed. */
export async function applyBulk(
  orgId: string,
  action: 'update' | 'delete',
  ids: string[],
  patch: Extract<BulkRequest, { action: 'update' }>['patch'] | undefined,
  opts: { signal?: AbortSignal; onProgress?: (done: number, total: number) => void } = {},
): Promise<ChunkedResult> {
  try {
    return await runChunked(
      ids,
      (chunk) => postBulk(orgId, action === 'delete' ? { action, ids: chunk } : { action, ids: chunk, patch: patch as NonNullable<typeof patch> }),
      opts,
    )
  } finally {
    notifyLedgerChanged()
  }
}

// lib/utils/journal-url.ts
import { EMPTY_FILTERS, filtersToParams, parseFilters, type JournalFilters } from '@/lib/domain/journal/filter'
import { fromDay, toDay } from '@/lib/domain/ledger/days'
import { resolvePeriod, type PeriodPreset, type PeriodSelection } from '@/lib/utils/period'

/** The journal's period menu: all time first (Finmap's default), the reports' presets after it. */
export const JOURNAL_PERIOD_PRESETS: readonly PeriodPreset[] = [
  'allTime',
  'currentMonth',
  'prevMonth',
  'currentWeek',
  'prevWeek',
  'currentYear',
  'prevYear',
  'custom',
]
export const DEFAULT_JOURNAL_PERIOD: PeriodSelection = { preset: 'allTime' }

export interface JournalView {
  filters: JournalFilters
  period: PeriodSelection
}

function daysOf(period: PeriodSelection, now: Date): { from?: string; to?: string } {
  const range = resolvePeriod(period, now)
  return { ...(range.start ? { from: toDay(range.start) } : {}), ...(range.end ? { to: toDay(range.end) } : {}) }
}

/**
 * The journal view from the address. A preset is stored by name and resolved against today, so a saved link
 * to «this month» stays this month; a custom range stores its two days. The account is the shell's selection
 * and never comes from the address.
 */
export function viewFromParams(params: URLSearchParams, now: Date): JournalView {
  const parsed = parseFilters(params)
  const { accountId: _account, from, to, ...rest } = parsed
  void _account
  const preset = params.get('period') as PeriodPreset | null

  if (preset === 'custom' && from && to) {
    return { filters: { ...rest, from, to }, period: { preset: 'custom', start: fromDay(from), end: fromDay(to) } }
  }
  if (preset && preset !== 'custom' && preset !== 'allTime' && JOURNAL_PERIOD_PRESETS.includes(preset)) {
    const period: PeriodSelection = { preset }
    return { filters: { ...rest, ...daysOf(period, now) }, period }
  }
  return { filters: { ...EMPTY_FILTERS, ...rest }, period: DEFAULT_JOURNAL_PERIOD }
}

export function viewToParams(view: JournalView): URLSearchParams {
  const { accountId: _account, from, to, ...rest } = view.filters
  void _account
  const custom = view.period.preset === 'custom'
  const params = filtersToParams({ ...rest, ...(custom ? { from, to } : {}) })
  if (view.period.preset !== 'allTime') params.set('period', view.period.preset)
  return params
}

// lib/domain/journal/paging.ts
import type { JournalPage, JournalRow } from './types'

export interface PagingState {
  /** The orgId and filters these rows belong to; an answer for another key is stale and ignored. */
  key: string
  rows: JournalRow[]
  next: string | null
  phase: 'loading' | 'ready' | 'error'
  moreLoading: boolean
}

export type PagingAction =
  | { type: 'first'; key: string; page: JournalPage }
  | { type: 'moreStart'; key: string }
  | { type: 'more'; key: string; page: JournalPage }
  | { type: 'replace'; key: string; rows: JournalRow[]; next: string | null }
  | { type: 'fail'; key: string }

export const initialPaging = (key: string): PagingState => ({ key, rows: [], next: null, phase: 'loading', moreLoading: false })

export function pagingReducer(state: PagingState, action: PagingAction): PagingState {
  if (action.key !== state.key) return state
  switch (action.type) {
    case 'first':
      return { key: state.key, rows: action.page.rows, next: action.page.next, phase: 'ready', moreLoading: false }
    case 'moreStart':
      return { ...state, moreLoading: true }
    case 'more':
      return { ...state, rows: [...state.rows, ...action.page.rows], next: action.page.next, moreLoading: false }
    case 'replace':
      return { ...state, rows: action.rows, next: action.next, phase: 'ready', moreLoading: false }
    case 'fail':
      return { ...state, phase: 'error', moreLoading: false }
  }
}

// lib/hooks/useJournal.ts
'use client'

import { useCallback, useEffect, useReducer, useRef } from 'react'
import { fetchJournalPage, PAGE_SIZE } from '@/lib/api/journal-client'
import { onLedgerChanged } from '@/lib/api/ledger-events'
import { filtersToParams, type JournalFilters } from '@/lib/domain/journal/filter'
import { initialPaging, pagingReducer } from '@/lib/domain/journal/paging'
import type { JournalRow } from '@/lib/domain/journal/types'

const RELOAD_DEBOUNCE_MS = 300
const MAX_RELOAD_PAGES = 20

export interface UseJournalResult {
  rows: JournalRow[]
  loading: boolean
  error: boolean
  hasMore: boolean
  loadingMore: boolean
  loadMore: () => void
}

/**
 * The journal pages for an org and filters. `filters` must keep its identity while its content does
 * (memoise it on the query string). Operations changed in this browser (forms, bulk actions, undo) reload
 * what is shown, to the same depth, without jumping back to the first page.
 */
export function useJournal(orgId: string | null, filters: JournalFilters): UseJournalResult {
  const key = `${orgId ?? ''}|${filtersToParams(filters).toString()}`
  const [state, dispatch] = useReducer(pagingReducer, key, initialPaging)
  const latest = useRef({ state, key })
  useEffect(() => {
    latest.current = { state, key }
  })

  // First page. Nothing is set synchronously: `loading` is derived from the key the rows belong to.
  useEffect(() => {
    if (!orgId) return
    const ac = new AbortController()
    fetchJournalPage(orgId, filters, null, { signal: ac.signal }).then(
      (page) => dispatch({ type: 'first', key, page }),
      () => {
        if (!ac.signal.aborted) dispatch({ type: 'fail', key })
      },
    )
    return () => ac.abort()
  }, [orgId, filters, key])

  const loadMore = useCallback(() => {
    const current = latest.current
    if (!orgId || current.key !== key || current.state.next === null || current.state.moreLoading || current.state.phase !== 'ready') return
    dispatch({ type: 'moreStart', key })
    fetchJournalPage(orgId, filters, current.state.next).then(
      (page) => dispatch({ type: 'more', key, page }),
      () => dispatch({ type: 'fail', key }),
    )
  }, [orgId, filters, key])

  // Reload to the depth the person has scrolled to, after this browser changed operations.
  useEffect(() => {
    if (!orgId) return
    let timer: ReturnType<typeof setTimeout> | undefined
    const reload = async () => {
      const { state: s, key: k } = latest.current
      if (k !== key || s.phase === 'loading') return
      const want = Math.max(s.rows.length, PAGE_SIZE)
      const rows: JournalRow[] = []
      let cursor: string | null = null
      for (let i = 0; i < MAX_RELOAD_PAGES; i++) {
        const page = await fetchJournalPage(orgId, filters, cursor)
        rows.push(...page.rows)
        cursor = page.next
        if (cursor === null || rows.length >= want) break
      }
      dispatch({ type: 'replace', key, rows, next: cursor })
    }
    const off = onLedgerChanged(() => {
      clearTimeout(timer)
      timer = setTimeout(() => void reload().catch(() => dispatch({ type: 'fail', key })), RELOAD_DEBOUNCE_MS)
    })
    return () => {
      off()
      clearTimeout(timer)
    }
  }, [orgId, filters, key])

  const current = state.key === key
  return {
    rows: current ? state.rows : [],
    loading: !current || state.phase === 'loading',
    error: current && state.phase === 'error',
    hasMore: current && state.next !== null,
    loadingMore: current && state.moreLoading,
    loadMore,
  }
}

const EMPTY: ReadonlySet<string> = new Set()

/** The checked rows. The selection belongs to a scope (the filters) and is empty again when the scope changes. */
export function useJournalSelection(scope: string) {
  const [state, setState] = useState<{ scope: string; ids: ReadonlySet<string> }>({ scope, ids: EMPTY })
  const ids = state.scope === scope ? state.ids : EMPTY
  const set = (next: ReadonlySet<string>) => setState({ scope, ids: next })
  return {
    ids,
    has: (id: string) => ids.has(id),
    toggle: (id: string) => {
      const next = new Set(ids)
      if (!next.delete(id)) next.add(id)
      set(next)
    },
    add: (list: string[]) => set(new Set([...ids, ...list])),
    remove: (list: string[]) => set(new Set([...ids].filter((id) => !list.includes(id)))),
    clear: () => set(EMPTY),
  }
}

// lib/api/journal-actions.ts
import { revertTransaction, restoreTransaction } from '@/lib/api/transaction-undo'
import { toUiTransaction } from '@/lib/firebase/ledger-map'
import type { BulkOutcome } from '@/lib/domain/journal/bulk-input'
import { splitByPatch, type BulkPatch, type SkipReason } from '@/lib/domain/journal/selection'
import type { LedgerTransaction } from '@/lib/domain/ledger/types'
import { applyBulk, type ChunkedResult } from './journal-bulk'

type Send = (action: 'update' | 'delete', ids: string[], patch?: BulkPatch) => Promise<BulkOutcome[]>

export interface BulkReport {
  /** Rows the server changed (or deleted): what an undo has to put back, as they were. */
  applied: LedgerTransaction[]
  failed: { id: string; code: string }[]
  skipped: { id: string; reason: SkipReason }[]
  stopped: boolean
}

interface Options {
  /** Warsaw today, YYYY-MM-DD. */
  today: string
  signal?: AbortSignal
  onProgress?: (done: number, total: number) => void
  /** For tests: replaces the network. */
  send?: Send
}

function report(rows: LedgerTransaction[], outcomes: BulkOutcome[], skipped: BulkReport['skipped'], stopped: boolean): BulkReport {
  const byId = new Map(rows.map((r) => [r.id, r]))
  const applied: LedgerTransaction[] = []
  const failed: BulkReport['failed'] = []
  for (const o of outcomes) {
    if (o.ok) applied.push(byId.get(o.id) as LedgerTransaction)
    else failed.push({ id: o.id, code: o.code })
  }
  return { applied, failed, skipped, stopped }
}

async function run(orgId: string | null, action: 'update' | 'delete', ids: string[], patch: BulkPatch | undefined, o: Options): Promise<ChunkedResult> {
  if (o.send) {
    const outcomes: BulkOutcome[] = []
    const send = o.send
    // The injected sender is called once with everything: chunking is covered by its own test.
    outcomes.push(...(await send(action, ids, patch)))
    return { outcomes, stopped: false }
  }
  return applyBulk(orgId as string, action, ids, patch, { signal: o.signal, onProgress: o.onProgress })
}

/** Applies `patch` to the selection (rows it cannot apply to are skipped before anything is sent). */
export async function bulkChange(rows: LedgerTransaction[], patch: BulkPatch, o: Options & { orgId?: string }): Promise<BulkReport> {
  const { apply, skipped } = splitByPatch(rows, patch, o.today)
  const { outcomes, stopped } = await run(o.orgId ?? null, 'update', apply.map((r) => r.id), patch, o)
  return report(apply, outcomes, skipped, stopped)
}

export async function bulkDelete(rows: LedgerTransaction[], o: Omit<Options, 'today'> & { orgId?: string }): Promise<BulkReport> {
  const { outcomes, stopped } = await run(o.orgId ?? null, 'delete', rows.map((r) => r.id), undefined, { today: '', ...o })
  return report(rows, outcomes, [], stopped)
}

interface UndoResult {
  done: number
  failed: number
}

async function each(rows: LedgerTransaction[], step: (row: LedgerTransaction) => Promise<unknown>): Promise<UndoResult> {
  let done = 0
  let failed = 0
  for (const row of rows) {
    try {
      await step(row)
      done++
    } catch {
      failed++
    }
  }
  return { done, failed }
}

/** Writes the rows back as they were before a bulk change (a full replace each, as the single-edit undo does). */
export function undoBulkChange(before: LedgerTransaction[], o: { orgId?: string; revert?: (row: LedgerTransaction) => Promise<unknown> } = {}): Promise<UndoResult> {
  const revert = o.revert ?? ((row: LedgerTransaction) => revertTransaction(o.orgId as string, toUiTransaction(o.orgId as string, row)))
  return each(before, revert)
}

/** Creates deleted rows again under their own ids. */
export function undoBulkDelete(rows: LedgerTransaction[], o: { orgId?: string; restore?: (row: LedgerTransaction) => Promise<unknown> } = {}): Promise<UndoResult> {
  const restore = o.restore ?? ((row: LedgerTransaction) => restoreTransaction(o.orgId as string, toUiTransaction(o.orgId as string, row)))
  return each(rows, restore)
}
```

              onChange={setMode}

                { value: 'date', label: t('accrualDate') },
                { value: 'period', label: t('accrualPeriod') },
              ]}
            />

                <DateField id="bulk-accrual-from" value={from} onChange={setFrom} {...dateProps} />
                <DateField id="bulk-accrual-to" value={to} onChange={setTo} {...dateProps} />
              </div>
            )}
          </div>
        )}

        {preview && (skippedTransfers > 0 || skippedFuture > 0) && (
          <p className="text-sm text-ink-muted">
            {t('preview', { apply: preview.apply.length, skipped: preview.skipped.length })}{' '}
            {skippedTransfers > 0 && t('skipTransfer', { count: skippedTransfers })} {skippedFuture > 0 && t('skipFuture', { count: skippedFuture })}
          </p>
        )}

        {progress && <p role="status" className="text-sm text-ink-muted tabular-nums">{t('progress', { done: progress.done, total: progress.total })}</p>}

        <DialogFooter>
          {busy ? (
            <Button type="button" variant="outline" onClick={onStop}>{t('stop')}</Button>
          ) : (
            <Button type="button" variant="outline" onClick={onClose}>{t('cancel')}</Button>
          )}
          <Button type="button" disabled={busy || !current || !preview || preview.apply.length === 0} onClick={() => current && onApply(current)}>
            {t('apply')}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )

export const exportFileName = (prefix: string, day: string): string => `${prefix}-${day}.csv`

/** Hands a text file to the browser's download. */
export function downloadText(filename: string, content: string, mime = 'text/csv;charset=utf-8'): void {
  const url = URL.createObjectURL(new Blob([content], { type: mime }))
  const link = document.createElement('a')
  link.href = url
  link.download = filename
  document.body.appendChild(link)
  link.click()
  document.body.removeChild(link)
  URL.revokeObjectURL(url)
}

'use client'

import { useState } from 'react'
import { useTranslations } from 'next-intl'
import { Download } from 'lucide-react'
import { toast } from 'sonner'
import { Button } from '@/components/ui/button'

import type { Counterparty } from '@/lib/domain/counterparties/types'

import type { JournalFilters } from '@/lib/domain/journal/filter'

import { useFormatters } from '@/lib/i18n/useFormatters'
import { downloadText, exportFileName } from '@/lib/utils/download'
import { todayDay } from '@/lib/utils/today'
import type { Account, Category } from '@/types'

/** Rows one file may hold; past that the person narrows the filter. */
const MAX_EXPORT_ROWS = 20_000
const EXPORT_PAGE = 100

interface Props {
  orgId: string
  filters: JournalFilters
  accounts: Account[]
  counterparties: Counterparty[]
  incomeCategories: Category[]
  expenseCategories: Category[]
  baseCurrency: string
}

export function ExportButton({ orgId, filters, accounts, counterparties, incomeCategories, expenseCategories, baseCurrency }: Props) {
  const t = useTranslations('journal.export')
  const { locale } = useFormatters()
  const categoryLabel = useCategoryName()
  const [count, setCount] = useState<number | null>(null)

  const run = async () => {
    const rows: JournalRow[] = []
    let cursor: string | null = null
    setCount(0)
    try {
      do {
        const page = await fetchJournalPage(orgId, filters, cursor, { limit: EXPORT_PAGE })
        rows.push(...page.rows)
        cursor = page.next
        setCount(rows.length)
        if (rows.length > MAX_EXPORT_ROWS) {
          toast.error(t('tooMany', { limit: MAX_EXPORT_ROWS }))
          return
        }
      } while (cursor)
      if (rows.length === 0) {
        toast.info(t('empty'))
        return
      }
      const accountName = new Map(accounts.map((a) => [a.id, a.name]))
      const counterpartyName = new Map(counterparties.map((c) => [c.id, c.name]))
      const categoryName = new Map([...incomeCategories, ...expenseCategories].map((c) => [c.id, categoryLabel(c)]))
      const ctx = {
        decimal: locale === 'en' ? ('.' as const) : (',' as const),
        accountName: (id: string) => accountName.get(id) ?? id,
        categoryName: (id: string) => categoryName.get(id) ?? id,
        counterpartyName: (id: string) => counterpartyName.get(id) ?? id,
        statusLabel: (s: string) => t(`status_${s}` as 'status_actual'),
        typeLabel: (type: string) => t(`type_${type}` as 'type_income'),
        baseCurrency,
      }
      const header = EXPORT_COLUMNS.map((c) => t(`column_${c}`))
      downloadText(exportFileName('journal', todayDay()), buildCsv(header, rows.map((r) => exportRow(r, ctx))))
    } catch {
      toast.error(t('failed'))
    } finally {
      setCount(null)
    }
  }

  return (
    <Button type="button" variant="outline" size="lg" onClick={() => void run()} disabled={count !== null} className="gap-1.5 border-line bg-surface text-ink-muted">
      <Download className="size-4" />
      {count === null ? t('button') : t('building', { count })}
    </Button>
  )
}
