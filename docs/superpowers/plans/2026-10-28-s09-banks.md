# S09 Banks Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Bank accounts from Enable Banking are linked to FinFlow accounts, bank operations land in the journal without duplicates, sync runs on app open and by a button, the balance matches the bank, and the owner is warned 3, 2, 1 days before the bank access ends.

**Architecture:** Server module `lib/server/banking/` (EB client behind an `EbApi` interface, linking, one-step sync with a lock and a cursor, server-only collections `bankAccounts` / `bankSeen`). All operations are written through the S05 ledger (`createTransaction` with a deterministic `clientId`), extended with a `bank` provenance block and a `bank_locked` guard. The browser drives a sync loop (one step = one EB page, at most 25 new operations) against `POST /api/orgs/{orgId}/bank/sync`. Pure domain code (`lib/domain/banking/`) holds the mapping, the dedup key input, the balance fit and the consent notice.

**Tech Stack:** Next.js 16 route handlers, firebase-admin (Firestore transactions), Enable Banking REST API, Vitest (unit + integration on the Firestore emulator), Playwright (Auth + Firestore emulators, a local fake bank), next-intl (uk/pl/en), sonner.

**Spec:** `docs/superpowers/specs/2026-10-28-s09-banks-design.md` (read it first). Working branch: `s09-banks` in `F:\AI\finflow`.

```ts
// Read-only probe of an existing Enable Banking connection (S09 spike). Prints the SHAPE of the answers only:
// key names, counts, presence rates, balance types. Never prints names, IBANs, amounts or references.
//   npm run spike:eb -- --project=finflow-e94b5 --org=<orgId>
// Needs .secrets/enablebanking-app-id.txt, .secrets/enablebanking-private.key and the Firebase service-account JSON.
import fs from 'node:fs'
import path from 'node:path'
import { cert, initializeApp } from 'firebase-admin/app'
import { getFirestore } from 'firebase-admin/firestore'
import { createJwt } from '@/lib/enablebanking/jwt'

const PROJECT_ID = 'finflow-e94b5'
const BASE = process.env.EB_API_BASE ?? 'https://api.enablebanking.com'

function arg(name: string): string | undefined {
  const hit = process.argv.find((a) => a.startsWith(`--${name}=`))
  return hit?.slice(name.length + 3)
}

function fail(message: string): never {
  console.error(message)
  process.exit(2)
}

/** Structure of a JSON value without its content. */
function shape(v: unknown, depth = 0): unknown {
  if (v === null) return 'null'
  if (Array.isArray(v)) return depth > 3 ? `array(${v.length})` : [`array(${v.length})`, v.length ? shape(v[0], depth + 1) : null]
  if (typeof v === 'object') {
    if (depth > 3) return 'object'
    return Object.fromEntries(Object.entries(v as Record<string, unknown>).map(([k, x]) => [k, shape(x, depth + 1)]))
  }
  if (typeof v === 'string') return `string(${v.length})`
  return typeof v
}

async function call(token: string, pathAndQuery: string): Promise<{ status: number; json: unknown }> {
  const res = await fetch(`${BASE}${pathAndQuery}`, { headers: { Authorization: `Bearer ${token}` } })
  const text = await res.text()
  let json: unknown = null
  try {
    json = JSON.parse(text)
  } catch {
    json = { nonJson: true, length: text.length }
  }
  return { status: res.status, json }
}

function isoDay(offsetDays: number): string {
  return new Date(Date.now() + offsetDays * 86_400_000).toISOString().slice(0, 10)
}

async function main() {
  if (arg('project') !== PROJECT_ID) fail(`Refused: pass --project=${PROJECT_ID} (production, read-only).`)
  const orgId = arg('org')
  if (!orgId) fail('Pass --org=<orgId>.')

  const secrets = path.resolve('.secrets')
  process.env.ENABLEBANKING_APP_ID = fs.readFileSync(path.join(secrets, 'enablebanking-app-id.txt'), 'utf8').trim()
  process.env.ENABLEBANKING_PRIVATE_KEY = fs.readFileSync(path.join(secrets, 'enablebanking-private.key'), 'utf8')
  const keyFile = fs.readdirSync(secrets).find((f) => /^finflow-.*-firebase-adminsdk-.*\.json$/.test(f))
  if (!keyFile) fail('No Firebase service-account JSON in .secrets/.')
  const app = initializeApp({ credential: cert(JSON.parse(fs.readFileSync(path.join(secrets, keyFile), 'utf8'))), projectId: PROJECT_ID })
  const db = getFirestore(app)

  const connections = await db.collection(`organizations/${orgId}/bankConnections`).get()
  console.log(`connections: ${connections.size}`)
  const token = await createJwt()

  for (const doc of connections.docs) {
    const sessionId = doc.get('sessionId') as string | undefined
    console.log(`\n== connection ${doc.id.slice(0, 6)}… institution=${String(doc.get('institutionName'))} connectedAt=${String(doc.get('connectedAt'))}`)
    if (!sessionId) {
      console.log('no sessionId on the document')
      continue
    }
    const session = await call(token, `/sessions/${encodeURIComponent(sessionId)}`)
    console.log('GET /sessions/{id}', session.status)
    console.log('session shape:', JSON.stringify(shape(session.json)))
    const body = session.json as Record<string, unknown>
    const access = (body?.access ?? {}) as Record<string, unknown>
    console.log('access.valid_until:', String(access.valid_until ?? 'absent'), '| psu_type:', String(body?.psu_type ?? 'absent'), '| status:', String(body?.status ?? 'absent'))

    const accounts = ((body?.accounts_data ?? body?.accounts) as unknown[]) ?? []
    for (const [i, raw] of accounts.entries()) {
      const acc = (typeof raw === 'string' ? { uid: raw } : raw) as Record<string, unknown>
      const uid = acc.uid as string | undefined
      console.log(`\n-- account #${i + 1}: keys=${Object.keys(acc).join(',')} currency=${String(acc.currency ?? 'absent')} hasIban=${JSON.stringify(acc.account_id ?? null) !== 'null'} hasIdentificationHash=${acc.identification_hash !== undefined}`)
      if (!uid) continue

      const details = await call(token, `/accounts/${encodeURIComponent(uid)}/details`)
      console.log('GET details', details.status, 'shape:', JSON.stringify(shape(details.json)))

      const balances = await call(token, `/accounts/${encodeURIComponent(uid)}/balances`)
      const list = ((balances.json as Record<string, unknown>)?.balances as Record<string, unknown>[]) ?? []
      console.log('GET balances', balances.status, 'types:', list.map((b) => `${String(b.balance_type)}/${String((b.balance_amount as Record<string, unknown>)?.currency)}`).join(','), 'shape:', JSON.stringify(shape(list[0] ?? null)))

      const from = isoDay(-90)
      let key: string | null = null
      let pages = 0
      let total = 0
      const presence: Record<string, number> = {}
      const statuses: Record<string, number> = {}
      const indicators: Record<string, number> = {}
      const fingerprints = new Map<string, number>()
      let minDay = '9999'
      let maxDay = '0000'
      let firstError: { status: number; json: unknown } | null = null
      do {
        const q = new URLSearchParams({ date_from: from, transaction_status: 'BOOK' })
        if (key) q.set('continuation_key', key)
        const page = await call(token, `/accounts/${encodeURIComponent(uid)}/transactions?${q}`)
        if (page.status !== 200) {
          firstError = page
          break
        }
        pages++
        const data = page.json as Record<string, unknown>
        const txs = (data.transactions as Record<string, unknown>[]) ?? []
        key = (data.continuation_key as string | null | undefined) ?? null
        if (pages === 1) console.log('transaction shape:', JSON.stringify(shape(txs[0] ?? null)))
        for (const t of txs) {
          total++
          for (const [k, v] of Object.entries(t)) if (v !== null && v !== undefined && v !== '') presence[k] = (presence[k] ?? 0) + 1
          statuses[String(t.status)] = (statuses[String(t.status)] ?? 0) + 1
          indicators[String(t.credit_debit_indicator)] = (indicators[String(t.credit_debit_indicator)] ?? 0) + 1
          const day = String(t.booking_date ?? t.value_date ?? t.transaction_date ?? '')
          if (day && day < minDay) minDay = day
          if (day && day > maxDay) maxDay = day
          const fp = JSON.stringify([day, (t.transaction_amount as Record<string, unknown>)?.amount, t.credit_debit_indicator, JSON.stringify(t.creditor ?? t.debtor ?? null), JSON.stringify(t.remittance_information ?? null)])
          fingerprints.set(fp, (fingerprints.get(fp) ?? 0) + 1)
        }
      } while (key && pages < 50)
      if (firstError) console.log('transactions error', firstError.status, JSON.stringify(shape(firstError.json)))
      const dupGroups = [...fingerprints.values()].filter((n) => n > 1).length
      console.log(`transactions: total=${total} pages=${pages} range=${minDay}..${maxDay} statuses=${JSON.stringify(statuses)} indicators=${JSON.stringify(indicators)}`)
      console.log('field presence:', JSON.stringify(presence))
      console.log(`identical-fingerprint groups (same day/amount/party/text): ${dupGroups}`)
    }
  }
}

main().catch((e) => {
  console.error('spike failed:', e instanceof Error ? e.message : String(e))
  process.exit(1)
})
```

```ts
import { LedgerError } from './errors'
import type { LedgerTransaction } from './types'

/** The fields that must keep matching the bank statement. */
const LOCKED = ['type', 'status', 'accountId', 'amountMinor', 'paymentDate'] as const

/** An operation imported from a bank may change everything except what the statement fixes. */
export function assertBankUnlocked(prior: LedgerTransaction, next: LedgerTransaction): void {
  for (const field of LOCKED) {
    if (prior[field] !== next[field]) throw new LedgerError('bank_locked', field)
  }
}
```

```ts
import { fromDay } from '@/lib/domain/ledger/days'

/** The notice starts this many days before the access ends. */
export const NOTICE_DAYS = 3

export type ConsentNotice =
  | { kind: 'none' }
  | { kind: 'days'; days: 1 | 2 | 3 }
  | { kind: 'today' }
  | { kind: 'expired' }

/** Both arguments are Warsaw calendar days (`YYYY-MM-DD`): the last day of the access and today. */
export function consentNotice(validUntilDay: string, today: string): ConsentNotice {
  const left = Math.round((fromDay(validUntilDay).getTime() - fromDay(today).getTime()) / 86_400_000)
  if (left < 0) return { kind: 'expired' }
  if (left === 0) return { kind: 'today' }
  if (left <= NOTICE_DAYS) return { kind: 'days', days: left as 1 | 2 | 3 }
  return { kind: 'none' }
}
```

```ts
import { isDay } from '@/lib/domain/ledger/days'
import type { BankInfo } from '@/lib/domain/ledger/types'

/** One operation as Enable Banking returns it (only the fields S09 reads). */
export interface EbTransaction {
  entry_reference?: string | null
  transaction_id?: string | null
  booking_date?: string | null
  value_date?: string | null
  transaction_date?: string | null
  transaction_amount: { amount: string; currency: string }
  credit_debit_indicator: 'CRDT' | 'DBIT'
  status?: string | null
  creditor?: { name?: string | null } | null
  debtor?: { name?: string | null } | null
  remittance_information?: string[] | null
  note?: string | null
}

export interface EbBalance {
  balance_type: string
  balance_amount: { amount: string; currency: string }
  credit_debit_indicator?: 'CRDT' | 'DBIT' | null
}

export interface MapContext {
  bankAccountId: string
  accountId: string
  accountCurrency: string
  /** The deterministic operation id (`bk_` + key). */
  clientId: string
}

export type MapResult =
  | { ok: true; input: Record<string, unknown>; bank: BankInfo }
  | { ok: false; reason: 'status' | 'currency' | 'amount' | 'date' }

/** "123.45" -> 12345. Exactly the shapes banks send for 2-decimal currencies; anything else is null. */
export function parseDecimalMinor(s: unknown): number | null {
  if (typeof s !== 'string') return null
  const m = /^(-?)(\d{1,12})(?:\.(\d{1,2}))?$/.exec(s.trim())
  if (!m) return null
  const minor = Number(m[2]) * 100 + Number((m[3] ?? '').padEnd(2, '0'))
  if (minor === 0) return 0
  return m[1] === '-' ? -minor : minor
}

function clean(text: string | null | undefined, max: number): string | undefined {
  if (typeof text !== 'string') return undefined
  const t = text.replace(/\s+/g, ' ').trim()
  return t.length === 0 ? undefined : t.slice(0, max)
}

/** Turns a booked bank operation into the input of `parseTransactionInput` plus the provenance block. */
export function mapBankTransaction(raw: EbTransaction, ctx: MapContext): MapResult {
  if (raw.status !== undefined && raw.status !== null && raw.status !== 'BOOK') return { ok: false, reason: 'status' }
  if (raw.transaction_amount?.currency !== ctx.accountCurrency) return { ok: false, reason: 'currency' }
  const signed = parseDecimalMinor(raw.transaction_amount?.amount)
  const amountMinor = signed === null ? 0 : Math.abs(signed)
  if (amountMinor === 0) return { ok: false, reason: 'amount' }
  const paymentDate = [raw.booking_date, raw.value_date, raw.transaction_date].find((d) => isDay(d))
  if (paymentDate === undefined) return { ok: false, reason: 'date' }

  const debit = raw.credit_debit_indicator === 'DBIT'
  const remittance = clean((raw.remittance_information ?? []).join(' '), 500)
  const description = remittance ?? clean(raw.note, 500)
  const counterpartyName = clean(debit ? raw.creditor?.name : raw.debtor?.name, 200)
  const entryReference = clean(raw.entry_reference ?? raw.transaction_id, 200)

  const input: Record<string, unknown> = {
    clientId: ctx.clientId,
    type: debit ? 'expense' : 'income',
    status: 'actual',
    accountId: ctx.accountId,
    amountMinor,
    paymentDate,
  }
  if (description !== undefined) input.description = description

  const bank: BankInfo = { accountRef: ctx.bankAccountId }
  if (entryReference !== undefined) bank.entryReference = entryReference
  if (counterpartyName !== undefined) bank.counterpartyName = counterpartyName
  if (remittance !== undefined) bank.remittance = remittance
  return { ok: true, input, bank }
}

const BALANCE_ORDER = ['CLBD', 'ITBD', 'XPCD', 'OPBD', 'VALU']

/** The booked balance to compare with the ledger: closing booked first, then interim booked, then any. */
export function pickBalance(balances: EbBalance[], currency: string): { amountMinor: number; type: string } | null {
  const usable = balances.filter((b) => b.balance_amount?.currency === currency && parseDecimalMinor(b.balance_amount.amount) !== null)
  if (usable.length === 0) return null
  const rank = (t: string) => {
    const i = BALANCE_ORDER.indexOf(t)
    return i === -1 ? BALANCE_ORDER.length : i
  }
  const best = [...usable].sort((a, b) => rank(a.balance_type) - rank(b.balance_type))[0]
  const magnitude = parseDecimalMinor(best.balance_amount.amount) as number
  const amountMinor = best.credit_debit_indicator === 'DBIT' ? -Math.abs(magnitude) : magnitude
  return { amountMinor, type: best.balance_type }
}
```

```ts
import { createHash } from 'node:crypto'
import type { EbTransaction } from '@/lib/domain/banking/map'

function norm(s: string | null | undefined): string {
  return (s ?? '').replace(/\s+/g, ' ').trim().toLowerCase()
}

function sha(parts: string[]): string {
  return createHash('sha256').update(parts.join('\u0001')).digest('hex').slice(0, 32)
}

/**
 * The identity of each bank operation: 32 hex characters, stable between reads. With a bank reference
 * (`entry_reference`, else `transaction_id`) it is that reference inside the bank account. Without one it is a
 * fingerprint of day, amount, direction, other party and text, plus the ordinal among identical operations of the
 * same read (two identical coffees on one day stay two operations). The ordinal is per read: identical twins that
 * a bank splits across pages could be keyed differently (documented risk, S09 spec).
 */
export function bankTxKeys(bankAccountId: string, txs: EbTransaction[]): Array<{ key: string; tx: EbTransaction }> {
  const seen = new Map<string, number>()
  return txs.map((tx) => {
    const ref = (tx.entry_reference ?? tx.transaction_id ?? '').trim()
    if (ref !== '') return { key: sha([bankAccountId, 'ref', ref]), tx }
    const party = tx.credit_debit_indicator === 'DBIT' ? tx.creditor?.name : tx.debtor?.name
    const fingerprint = [
      bankAccountId,
      'fp',
      tx.booking_date ?? tx.value_date ?? tx.transaction_date ?? '',
      tx.transaction_amount?.amount ?? '',
      tx.transaction_amount?.currency ?? '',
      tx.credit_debit_indicator,
      norm(party),
      norm((tx.remittance_information ?? []).join(' ')),
    ]
    const id = fingerprint.join('\u0001')
    const ordinal = seen.get(id) ?? 0
    seen.set(id, ordinal + 1)
    return { key: sha([...fingerprint, String(ordinal)]), tx }
  })
}
```

```ts
/**
 * The code of a failed Enable Banking answer. The patterns are a first guess from the API documentation;
 * refine them with what the real bank answers (see the spike results in the spec).
 */
export function classifyEbFailure(status: number, body: string): BankErrorCode {
  const b = body.toLowerCase()
  if (status === 429 || /rate.?limit|too many/.test(b)) return 'rate_limited'
  // Our own credentials are wrong: not the person's consent.
  if (/jwt|signature|\bkid\b/.test(b)) return 'bank_error'
  if ([401, 403, 404, 422].includes(status) && /session|consent|expired|closed|revoked/.test(b)) return 'consent_expired'
  if (status >= 500) return 'bank_unavailable'
  return 'bank_error'
}

function str(v: unknown): string | null {
  return typeof v === 'string' && v !== '' ? v : null
}

function rec(v: unknown): Record<string, unknown> | null {
  return typeof v === 'object' && v !== null && !Array.isArray(v) ? (v as Record<string, unknown>) : null
}

function normalizeAccount(raw: unknown): EbAccount | null {
  if (typeof raw === 'string') return raw === '' ? null : { uid: raw, identificationHash: null, iban: null, currency: null, name: null }
  const a = rec(raw)
  const uid = a ? str(a.uid) : null
  if (!a || !uid) return null
  return {
    uid,
    identificationHash: str(a.identification_hash),
    iban: str(rec(a.account_id)?.iban),
    currency: str(a.currency),
    name: str(a.name),
  }
}

export function normalizeSession(sessionId: string, json: unknown): EbSession {
  const body = rec(json) ?? {}
  const list = Array.isArray(body.accounts_data) && body.accounts_data.length > 0 ? body.accounts_data : Array.isArray(body.accounts) ? body.accounts : []
  const validUntil = str(rec(body.access)?.valid_until)
  return {
    sessionId,
    institutionName: str(rec(body.aspsp)?.name) ?? '',
    institutionCountry: str(rec(body.aspsp)?.country),
    psuType: body.psu_type === 'business' || body.psu_type === 'personal' ? body.psu_type : null,
    // We asked for 90 days; use that when the bank's answer does not say.
    validUntil: validUntil !== null && !Number.isNaN(Date.parse(validUntil)) ? validUntil : new Date(Date.now() + 90 * 86_400_000).toISOString(),
    accounts: list.map(normalizeAccount).filter((a): a is EbAccount => a !== null),
  }
}

function sanitize(text: string): string {
  return text.replace(/[\u0000-\u001f\u007f-\u009f]/g, ' ').slice(0, 200)
}

export function createEbApi(opts: EbApiOptions): EbApi {
  const base = opts.base ?? process.env.EB_API_BASE ?? 'https://api.enablebanking.com'
  const doFetch = opts.fetchImpl ?? fetch

  async function get(pathAndQuery: string): Promise<unknown> {
    const headers: Record<string, string> = { Authorization: `Bearer ${await opts.token()}` }
    if (opts.psu?.ip) headers['psu-ip-address'] = opts.psu.ip
    if (opts.psu?.userAgent) headers['psu-user-agent'] = opts.psu.userAgent
    let res: Response
    try {
      res = await doFetch(`${base}${pathAndQuery}`, { headers })
    } catch {
      throw new BankError('bank_unavailable')
    }
    if (!res.ok) {
      const text = await res.text().catch(() => '')
      console.error('Enable Banking call failed:', res.status, sanitize(text))
      throw new BankError(classifyEbFailure(res.status, text))
    }
    return res.json().catch(() => null)
  }

  return {
    async getSession(sessionId) {
      return normalizeSession(sessionId, await get(`/sessions/${encodeURIComponent(sessionId)}`))
    },
    async getBalances(accountUid) {
      const body = rec(await get(`/accounts/${encodeURIComponent(accountUid)}/balances`))
      return Array.isArray(body?.balances) ? (body.balances as EbBalance[]) : []
    },
    async getTransactions(accountUid, query) {
      const q = new URLSearchParams({ date_from: query.dateFrom, transaction_status: 'BOOK' })
      if (query.continuationKey) q.set('continuation_key', query.continuationKey)
      const body = rec(await get(`/accounts/${encodeURIComponent(accountUid)}/transactions?${q}`))
      return {
        transactions: Array.isArray(body?.transactions) ? (body.transactions as EbTransaction[]) : [],
        continuationKey: str(body?.continuation_key),
      }
    },
  }
}
```

```ts
/** Validates the body of POST /api/orgs/{orgId}/bank/accounts. Throws `LedgerError('invalid_input', field)`-shaped errors via `bad`. */
export function parseLinkRequest(body: unknown): LinkRequest {
  const r = asRecord(body)
  onlyKeys(r, ['connectionId', 'ebAccountUid', 'mode', 'accountId', 'name', 'importFrom'])
  if (!isId(r.connectionId)) bad('connectionId')
  if (typeof r.ebAccountUid !== 'string' || r.ebAccountUid.length < 1 || r.ebAccountUid.length > 200) bad('ebAccountUid')
  if (r.mode !== 'create' && r.mode !== 'existing') bad('mode')
  if (!isDay(r.importFrom) || r.importFrom < MIN_DAY) bad('importFrom')
  let accountId: string | undefined
  if (r.mode === 'existing') {
    if (!isId(r.accountId)) bad('accountId')
    accountId = r.accountId
  } else if (r.accountId !== undefined) {
    bad('accountId')
  }
  let name: string | undefined
  if (r.name !== undefined) {
    if (typeof r.name !== 'string') bad('name')
    const text = r.name.trim()
    if (text.length > 120) bad('name')
    if (text.length > 0) name = text
  }
  return { connectionId: r.connectionId, ebAccountUid: r.ebAccountUid, mode: r.mode, accountId, name, importFrom: r.importFrom }
}
```

```ts
import { parseAccount } from '@/lib/domain/ledger/doc'

import { BankError } from './errors'
import { bankAccountsCol, bankConnectionsCol } from './paths'
import { parseBankAccount, parseConnection } from './types'

const ACCOUNT_COLOR = '#00c896'

/**
 * Saves a bank session after the OAuth callback (server side, owner only). Bank accounts this company linked
 * before (same identification hash, else same IBAN) are re-pointed to the new session: the link, the start day
 * and every `bankSeen` mark stay, so renewing the access never duplicates anything.
 */
export async function saveSession(deps: BankDeps, orgId: string, uid: string, session: EbSession): Promise<{ connectionId: string; relinked: number; unlinked: number }> {
  await bankConnectionsCol(deps.db, orgId).doc(session.sessionId).set({
    orgId,
    sessionId: session.sessionId,
    institutionName: session.institutionName,
    institutionCountry: session.institutionCountry,
    psuType: session.psuType,
    connectedAt: deps.now().toISOString(),
    validUntil: session.validUntil,
    status: 'active',
    connectedBy: uid,
  })

  const existing = await bankAccountsCol(deps.db, orgId).get()
  const oldConnections = new Set<string>()
  let relinked = 0
  for (const acc of session.accounts) {
    const hit = existing.docs.find((d) => {
      const b = parseBankAccount(d.data())
      if (!b) return false
      if (acc.identificationHash !== null && b.identificationHash !== null) return b.identificationHash === acc.identificationHash
      return acc.iban !== null && b.iban !== null && b.iban === acc.iban
    })
    if (!hit) continue
    const before = parseBankAccount(hit.data()) as NonNullable<ReturnType<typeof parseBankAccount>>
    if (before.connectionId !== session.sessionId) oldConnections.add(before.connectionId)
    await hit.ref.update({ connectionId: session.sessionId, ebAccountUid: acc.uid, status: 'active', cursor: null, syncLock: null })
    relinked++
  }

  // An old connection nothing refers to any more disappears.
  for (const oldId of oldConnections) {
    const stillUsed = await bankAccountsCol(deps.db, orgId).where('connectionId', '==', oldId).limit(1).get()
    if (stillUsed.empty) await bankConnectionsCol(deps.db, orgId).doc(oldId).delete()
  }
  return { connectionId: session.sessionId, relinked, unlinked: session.accounts.length - relinked }
}

async function loadConnection(deps: BankDeps, orgId: string, connectionId: string) {
  const snap = await bankConnectionsCol(deps.db, orgId).doc(connectionId).get()
  const conn = snap.exists ? parseConnection(snap.data()) : null
  if (!conn) throw new BankError('not_found')
  return conn
}

/** The bank's accounts of a connection, each flagged with whether it is already linked. */
export async function listConnectionAccounts(deps: BankDeps, orgId: string, connectionId: string): Promise<{ accounts: Array<EbAccount & { linked: boolean }> }> {
  const conn = await loadConnection(deps, orgId, connectionId)
  const session = await deps.eb.getSession(conn.sessionId)
  const linked = new Set<string>()
  for (const d of (await bankAccountsCol(deps.db, orgId).where('connectionId', '==', connectionId).get()).docs) {
    const b = parseBankAccount(d.data())
    if (b && b.status === 'active') linked.add(b.ebAccountUid)
  }
  return { accounts: session.accounts.map((a) => ({ ...a, linked: linked.has(a.uid) })) }
}

export async function linkAccount(deps: BankDeps, orgId: string, uid: string, input: LinkRequest): Promise<{ bankAccountId: string; accountId: string }> {
  if (!isDay(input.importFrom) || input.importFrom < MIN_DAY || input.importFrom > deps.today()) throw new BankError('invalid_input')
  const conn = await loadConnection(deps, orgId, input.connectionId)
  if (conn.status === 'expired' || todayDay(new Date(conn.validUntil)) < deps.today()) throw new BankError('consent_expired')

  const session = await deps.eb.getSession(conn.sessionId)
  const bank = session.accounts.find((a) => a.uid === input.ebAccountUid)
  if (!bank) throw new BankError('not_found')
  const currency = bank.currency
  if (currency === null || !(LEDGER_CURRENCIES as readonly string[]).includes(currency)) throw new BankError('currency_unsupported')

  const linked = await bankAccountsCol(deps.db, orgId).get()
  for (const d of linked.docs) {
    const b = parseBankAccount(d.data())
    if (!b || b.status !== 'active') continue
    const sameBankAccount = (b.connectionId === input.connectionId && b.ebAccountUid === input.ebAccountUid) || (bank.identificationHash !== null && b.identificationHash === bank.identificationHash)
    if (sameBankAccount || (input.mode === 'existing' && b.accountId === input.accountId)) throw new BankError('already_linked')
  }

  let accountId: string
  if (input.mode === 'existing') {
    const snap = await accountsCol(deps.db, orgId).doc(input.accountId as string).get()
    const account = snap.exists ? parseAccount(snap.id, snap.data()) : null
    if (!account) throw new BankError('not_found')
    if (account.currency !== currency) throw new BankError('currency_mismatch')
    accountId = account.id
  } else {
    const name = input.name ?? `${conn.institutionName} ${currency}`.trim()
    accountId = (await createAccount(deps.ledger, orgId, uid, { name, type: 'bank', currency, color: ACCOUNT_COLOR, openingMinor: 0 })).id
  }

  const ref = bankAccountsCol(deps.db, orgId).doc()
  await ref.set({
    orgId,
    connectionId: input.connectionId,
    ebAccountUid: bank.uid,
    identificationHash: bank.identificationHash,
    iban: bank.iban,
    currency,
    accountId,
    importFrom: input.importFrom,
    status: 'active',
    lastSyncAt: null,
    syncedThrough: null,
    cursor: null,
    syncLock: null,
    bankBalance: null,
    firstSyncDone: false,
  })
  return { bankAccountId: ref.id, accountId }
}

/** Drops the connection. The bank accounts stay (history, `bankSeen`), inactive, so reconnecting relinks them. */
export async function disconnect(deps: BankDeps, orgId: string, connectionId: string): Promise<void> {
  const accounts = await bankAccountsCol(deps.db, orgId).where('connectionId', '==', connectionId).get()
  const batch = deps.db.batch()
  for (const d of accounts.docs) batch.update(d.ref, { status: 'disconnected', syncLock: null, cursor: null })
  batch.delete(bankConnectionsCol(deps.db, orgId).doc(connectionId))
  await batch.commit()
}
```

```ts
/** Ledger refusals that retrying cannot fix: the operation is marked seen with the reason and left out. */
const PERMANENT: ReadonlySet<LedgerErrorCode> = new Set(['future_actual', 'currency_mismatch', 'account_not_found', 'invalid_input'])

async function acquire(deps: BankDeps, ref: DocumentReference, token: string): Promise<BankAccount> {
  return deps.db.runTransaction(async (t) => {
    const snap = await t.get(ref)
    const acc = snap.exists ? parseBankAccount(snap.data()) : null
    if (!acc || acc.status !== 'active') throw new BankError('not_found')
    const nowMs = deps.now().getTime()
    if (acc.syncLock !== null && acc.syncLock.until > nowMs) throw new BankError('already_running')
    t.update(ref, { syncLock: { until: nowMs + LOCK_MS, token } })
    return acc
  })
}

async function release(deps: BankDeps, ref: DocumentReference, token: string): Promise<void> {
  await deps.db.runTransaction(async (t) => {
    const snap = await t.get(ref)
    const lock = snap.exists ? parseBankAccount(snap.data())?.syncLock : null
    if (lock && lock.token === token) t.update(ref, { syncLock: null })
  })
}

async function markExpired(deps: BankDeps, orgId: string, connectionId: string): Promise<void> {
  await bankConnectionsCol(deps.db, orgId).doc(connectionId).set({ status: 'expired' }, { merge: true })
}

async function assertConsent(deps: BankDeps, orgId: string, acc: BankAccount): Promise<void> {
  const snap = await bankConnectionsCol(deps.db, orgId).doc(acc.connectionId).get()
  const conn = snap.exists ? parseConnection(snap.data()) : null
  if (!conn) throw new BankError('not_found')
  if (conn.status === 'expired' || todayDay(new Date(conn.validUntil)) < deps.today()) {
    await markExpired(deps, orgId, acc.connectionId)
    throw new BankError('consent_expired')
  }
}

function startDay(acc: BankAccount): string {
  if (acc.syncedThrough === null) return acc.importFrom
  const from = addDays(acc.syncedThrough, -OVERLAP_DAYS)
  return from > acc.importFrom ? from : acc.importFrom
}

/**
 * One step of a sync: one bank page, at most MAX_NEW_PER_STEP new operations. `done: false` means "call again".
 * Safe to repeat and to interrupt: every operation has a deterministic id and a `bankSeen` mark, so a step that
 * died halfway only costs the next one a few skipped reads. The operation is written before its mark; a crash
 * between the two is healed by the replay of `createTransaction` (same id) on the next run.
 */
export async function syncStep(deps: BankDeps, orgId: string, uid: string, bankAccountId: string): Promise<SyncResult> {
  const ref = bankAccountsCol(deps.db, orgId).doc(bankAccountId)
  const token = randomUUID()
  const acc = await acquire(deps, ref, token)
  try {
    await assertConsent(deps, orgId, acc)

    const cursor: BankCursor = acc.cursor ?? { from: startDay(acc), continuationKey: null, startedOn: deps.today() }
    let page: EbTxPage
    try {
      page = await deps.eb.getTransactions(acc.ebAccountUid, { dateFrom: cursor.from, continuationKey: cursor.continuationKey })
    } catch (e) {
      if (e instanceof BankError && e.code === 'consent_expired') await markExpired(deps, orgId, acc.connectionId)
      throw e
    }

    const entries = bankTxKeys(bankAccountId, page.transactions)
    const seenCol = bankSeenCol(deps.db, orgId)
    const seenSnaps = entries.length === 0 ? [] : await deps.db.getAll(...entries.map((e) => seenCol.doc(e.key)))
    const seen = new Set(seenSnaps.filter((s) => s.exists).map((s) => s.id))

    let imported = 0
    let skipped = 0
    let failed = 0
    let stoppedEarly = false
    for (const { key, tx } of entries) {
      if (seen.has(key)) {
        skipped++
        continue
      }
      if (imported >= MAX_NEW_PER_STEP) {
        stoppedEarly = true
        break
      }
      const clientId = `bk_${key}`
      const mapped = mapBankTransaction(tx, { bankAccountId, accountId: acc.accountId, accountCurrency: acc.currency, clientId })
      if (!mapped.ok) {
        // A pending operation is not an operation yet: no mark, it will arrive booked.
        if (mapped.reason === 'status') {
          skipped++
          continue
        }
        await seenCol.doc(key).set({ txId: clientId, at: deps.now().toISOString(), reason: mapped.reason })
        failed++
        continue
      }
      try {
        await createTransaction(deps.ledger, orgId, uid, parseTransactionInput(mapped.input), mapped.bank)
      } catch (e) {
        if (e instanceof LedgerError && PERMANENT.has(e.code)) {
          await seenCol.doc(key).set({ txId: clientId, at: deps.now().toISOString(), reason: e.code })
        } else if (!(e instanceof LedgerError)) {
          throw e
        }
        failed++
        continue
      }
      await seenCol.doc(key).set({ txId: clientId, at: deps.now().toISOString() })
      imported++
    }

    const progress = { imported, skipped, failed, bankBalanceMinor: null, differenceMinor: null }
    if (stoppedEarly) {
      await ref.update({ cursor })
      return { ...progress, done: false }
    }
    if (page.continuationKey !== null) {
      await ref.update({ cursor: { ...cursor, continuationKey: page.continuationKey } })
      return { ...progress, done: false }
    }
    return { ...progress, ...(await finish(deps, orgId, uid, ref, acc, cursor)), done: true }
  } finally {
    await release(deps, ref, token)
  }
}

/** The last page is in: read the bank balance, fit the opening balance on the first run, close the run. */
async function finish(deps: BankDeps, orgId: string, uid: string, ref: DocumentReference, acc: BankAccount, cursor: BankCursor) {
  let bankBalance = acc.bankBalance
  try {
    const picked = pickBalance(await deps.eb.getBalances(acc.ebAccountUid), acc.currency)
    if (picked) bankBalance = { ...picked, at: deps.now().toISOString() }
  } catch (e) {
    if (e instanceof BankError && e.code === 'consent_expired') {
      await markExpired(deps, orgId, acc.connectionId)
      throw e
    }
    // Any other balance failure keeps the previous balance; the operations are in.
  }

  const update: Record<string, unknown> = { cursor: null, lastSyncAt: deps.now().toISOString(), syncedThrough: cursor.startedOn }
  if (bankBalance !== null) update.bankBalance = bankBalance
  const accountRef = accountsCol(deps.db, orgId).doc(acc.accountId)

  if (!acc.firstSyncDone && bankBalance !== null) {
    const snap = await accountRef.get()
    const account = snap.exists ? parseAccount(snap.id, snap.data()) : null
    if (account) {
      const opening = fittedOpening(bankBalance.amountMinor, account.balanceMinor, account.openingMinor)
      if (Math.abs(opening) <= MAX_MINOR) {
        await updateAccount(deps.ledger, orgId, uid, acc.accountId, { openingMinor: opening })
        update.firstSyncDone = true
      }
    }
  }
  await ref.update(update)

  if (bankBalance === null) return { bankBalanceMinor: null, differenceMinor: null }
  const after = await accountRef.get()
  const account = after.exists ? parseAccount(after.id, after.data()) : null
  return { bankBalanceMinor: bankBalance.amountMinor, differenceMinor: account ? bankDifference(bankBalance.amountMinor, account.balanceMinor) : null }
}
```

```ts
import type { BankAccountStatus, BankConnectionStatus, BankStatus } from '@/lib/domain/banking/status-types'
import type { Role } from '@/lib/shell/permissions'

import type { BankDeps } from './deps'

export async function readBankStatus(deps: BankDeps, orgId: string, role: Role): Promise<BankStatus> {
  const today = deps.today()
  const connections: BankConnectionStatus[] = []
  for (const d of (await bankConnectionsCol(deps.db, orgId).get()).docs) {
    const c = parseConnection(d.data())
    if (!c) continue
    const expired = c.status === 'expired' || todayDay(new Date(c.validUntil)) < today
    connections.push({
      id: d.id,
      institutionName: c.institutionName,
      institutionCountry: c.institutionCountry,
      psuType: c.psuType,
      status: expired ? 'expired' : 'active',
      validUntil: c.validUntil,
      connectedAt: c.connectedAt,
    })
  }
  connections.sort((a, b) => a.connectedAt.localeCompare(b.connectedAt))
  if (role !== 'owner') return { connections }

  const accounts: BankAccountStatus[] = []
  for (const d of (await bankAccountsCol(deps.db, orgId).get()).docs) {
    const b = parseBankAccount(d.data())
    if (!b) continue
    accounts.push({
      id: d.id,
      connectionId: b.connectionId,
      ibanTail: b.iban === null ? null : b.iban.slice(-4),
      currency: b.currency,
      accountId: b.accountId,
      importFrom: b.importFrom,
      lastSyncAt: b.lastSyncAt,
      bankBalanceMinor: b.bankBalance?.amountMinor ?? null,
      status: b.status,
    })
  }
  return { connections, accounts }
}
```

```ts
export interface NoticeConnection {
  id: string
  institutionName: string
  /** The Warsaw day the access ends. */
  validUntilDay: string
  status: 'active' | 'expired'
}

/** A notice that has something to say. */
export type ActiveNotice = Exclude<ConsentNotice, { kind: 'none' }>

const SEVERITY = (n: ActiveNotice): number => (n.kind === 'expired' ? 100 : n.kind === 'today' ? 50 : 10 - n.days)

/** The connection that most needs a message and the message; null when none does. */
export function mostUrgentNotice(connections: NoticeConnection[], today: string): { connection: NoticeConnection; notice: ActiveNotice } | null {
  let best: { connection: NoticeConnection; notice: ActiveNotice } | null = null
  for (const connection of connections) {
    const notice: ConsentNotice = connection.status === 'expired' ? { kind: 'expired' } : consentNotice(connection.validUntilDay, today)
    if (notice.kind === 'none') continue
    if (best === null || SEVERITY(notice) > SEVERITY(best.notice)) best = { connection, notice }
  }
  return best
}
```

```ts
import type { LinkRequest } from '@/lib/domain/banking/link-input'
import type { BankSessionAccount, BankStatus } from '@/lib/domain/banking/status-types'
import { notifyLedgerChanged } from './ledger-events'
import { readApiError } from './ledger-client'

const EVENT = 'finflow:bank-changed'
/** A sync step moves at most 25 operations or one bank page; this is only a stop against a runaway loop. */
const MAX_STEPS = 200

/** Called when the bank data changed (a sync ended, an account was linked, a connection dropped). */
export function notifyBankChanged(): void {
  if (typeof window !== 'undefined') window.dispatchEvent(new Event(EVENT))
}

export function onBankChanged(callback: () => void): () => void {
  window.addEventListener(EVENT, callback)
  return () => window.removeEventListener(EVENT, callback)
}

const base = (orgId: string) => `/api/orgs/${encodeURIComponent(orgId)}/bank`

export async function fetchBankStatus(orgId: string, signal?: AbortSignal): Promise<BankStatus> {
  const res = await fetch(base(orgId), { credentials: 'same-origin', signal })
  if (!res.ok) throw await readApiError(res)
  return (await res.json()) as BankStatus
}

export async function fetchSessionAccounts(orgId: string, connectionId: string): Promise<BankSessionAccount[]> {
  const res = await fetch(`${base(orgId)}/connections/${encodeURIComponent(connectionId)}`, { credentials: 'same-origin' })
  if (!res.ok) throw await readApiError(res)
  return ((await res.json()) as { accounts: BankSessionAccount[] }).accounts
}

export async function linkBankAccount(orgId: string, body: LinkRequest): Promise<{ bankAccountId: string; accountId: string }> {
  const res = await fetch(`${base(orgId)}/accounts`, {
    method: 'POST',
    credentials: 'same-origin',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  })
  if (!res.ok) throw await readApiError(res)
  notifyBankChanged()
  return (await res.json()) as { bankAccountId: string; accountId: string }
}

export async function disconnectBank(orgId: string, connectionId: string): Promise<void> {
  const res = await fetch(`${base(orgId)}/connections/${encodeURIComponent(connectionId)}`, { method: 'DELETE', credentials: 'same-origin' })
  if (!res.ok) throw await readApiError(res)
  notifyBankChanged()
}

export interface SyncTotals {
  imported: number
  skipped: number
  failed: number
  bankBalanceMinor: number | null
  differenceMinor: number | null
  /** Another tab or request is already syncing this account. */
  busy: boolean
}

interface StepAnswer {
  imported: number
  skipped: number
  failed: number
  done: boolean
  bankBalanceMinor: number | null
  differenceMinor: number | null
}

/**
 * Runs the sync of one bank account to the end: one request per step while the server says `done: false`.
 * Tells the journal to reload when operations arrived. Throws `LedgerApiError` (status, code) on a failure;
 * `already_running` is not a failure, it returns `busy: true`.
 */
export async function syncBankAccount(
  orgId: string,
  bankAccountId: string,
  opts: { signal?: AbortSignal; onProgress?: (totals: SyncTotals) => void } = {},
): Promise<SyncTotals> {
  const totals: SyncTotals = { imported: 0, skipped: 0, failed: 0, bankBalanceMinor: null, differenceMinor: null, busy: false }
  try {
    for (let step = 0; step < MAX_STEPS; step++) {
      const res = await fetch(`${base(orgId)}/sync`, {
        method: 'POST',
        credentials: 'same-origin',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ bankAccountId }),
        signal: opts.signal,
      })
      if (!res.ok) {
        const error = await readApiError(res)
        if (error.code === 'already_running') {
          totals.busy = true
          break
        }
        throw error
      }
      const answer = (await res.json()) as StepAnswer
      totals.imported += answer.imported
      totals.skipped += answer.skipped
      totals.failed += answer.failed
      if (answer.done) {
        totals.bankBalanceMinor = answer.bankBalanceMinor
        totals.differenceMinor = answer.differenceMinor
        break
      }
      opts.onProgress?.({ ...totals })
    }
  } finally {
    if (totals.imported > 0) notifyLedgerChanged()
    notifyBankChanged()
  }
  return totals
}
```

```ts
'use client'

import { useCallback, useEffect, useState } from 'react'
import { fetchBankStatus, onBankChanged } from '@/lib/api/bank-client'
import type { BankStatus } from '@/lib/domain/banking/status-types'

interface State {
  status: BankStatus | null
  loading: boolean
  error: boolean
}

/** The company's bank connections, read from the server; reloads when the tab comes back and when bank data changed. */
export function useBankStatus(orgId: string, enabled = true) {
  const [state, setState] = useState<State>({ status: null, loading: enabled, error: false })
  const [tick, setTick] = useState(0)
  const refresh = useCallback(() => setTick((n) => n + 1), [])

  useEffect(() => {
    if (!enabled) return
    const controller = new AbortController()
    fetchBankStatus(orgId, controller.signal).then(
      (status) => setState({ status, loading: false, error: false }),
      () => {
        if (!controller.signal.aborted) setState((s) => ({ ...s, loading: false, error: true }))
      },
    )
    return () => controller.abort()
  }, [orgId, enabled, tick])

  useEffect(() => onBankChanged(refresh), [refresh])
  useEffect(() => {
    const onVisible = () => {
      if (document.visibilityState === 'visible') refresh()
    }
    document.addEventListener('visibilitychange', onVisible)
    return () => document.removeEventListener('visibilitychange', onVisible)
  }, [refresh])

  return { ...state, refresh }
}
```

```tsx
'use client'

import { useEffect, useRef } from 'react'
import { useTranslations } from 'next-intl'
import { toast } from 'sonner'
import { syncBankAccount } from '@/lib/api/bank-client'
import { useBankStatus } from '@/lib/hooks/useBankStatus'
import { useShell } from '@/lib/hooks/useShell'

const STALE_MS = 60 * 60 * 1000

/**
 * Owner only: when the app opens, syncs every linked bank account whose last sync is older than an hour
 * (or never happened). Once per account per page load. A failure is silent: the expiry banner and the
 * settings page show what is wrong; the button there retries with a message.
 */
export function BankAutoSync() {
  const t = useTranslations('shell.bankSync')
  const { orgId, can } = useShell()
  const owner = can('bank.manage')
  const { status, refresh } = useBankStatus(orgId, owner)
  const attempted = useRef(new Set<string>())

  useEffect(() => {
    if (!owner || !status?.accounts) return
    const live = new Set(status.connections.filter((c) => c.status === 'active').map((c) => c.id))
    const due = status.accounts.filter(
      (a) => a.status === 'active' && live.has(a.connectionId) && !attempted.current.has(a.id) &&
        (a.lastSyncAt === null || Date.now() - Date.parse(a.lastSyncAt) > STALE_MS),
    )
    if (due.length === 0) return
    for (const a of due) attempted.current.add(a.id)
    // Not cancelled on cleanup: React Strict Mode runs the effect twice in development and the second run would
    // find the accounts already marked as attempted. A sync that outlives the page is harmless (the server locks).
    void (async () => {
      let imported = 0
      for (const a of due) {
        try {
          imported += (await syncBankAccount(orgId, a.id)).imported
        } catch {
          // See the component comment.
        }
      }
      if (imported > 0) toast.success(t('autoSynced', { count: imported }))
      refresh()
    })()
  }, [owner, status, orgId, refresh, t])

  return null
}
```
